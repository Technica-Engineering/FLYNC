46def _with_source(err: PydanticCustomError, locator: str) -> PydanticCustomError: 

47 """Return *err* re-created with ``yaml_path=locator`` added to its context.""" 

48 

49 ctx = dict(err.context or {}) 

50 if "yaml_path" in ctx: 

51 return err 

52 ctx["yaml_path"] = locator 

53 # err.type / err.message_template are typed as ``str`` but were originally constructed 

54 # from ``LiteralString``, so re-forwarding them through PydanticCustomError is safe. 

55 return PydanticCustomError(err.type, err.message_template, ctx)

197def _resolve_j1939_frame_ref( 

198 ref: J1939FrameRef, 

199 frames_on_bus: dict[str, J1939Frame], 

200) -> list[J1939Frame]: 

201 """Resolve *ref* against *frames_on_bus* and return every matching frame. 

202 

203 A reference may address its frame three ways: 

204 

205 * by ``frame_ref`` (name) — the named frame must exist on the bus (at most one match); 

206 * by ``can_id`` — the full on-bus identifier has its Source Address bits stripped and the remainder 

207 (priority + PGN base, ``can_id & ~0xFF``) is matched against each frame's derived ``can_id``. Several 

208 frames on a bus may share the same derived identifier (same PGN / priority), so the caller must treat 

209 more than one match as ambiguous rather than pick one; 

210 * by both — the two must agree on the same frame (double verification); the caller checks the *named* 

211 frame's derived ``can_id`` against the reference's base. 

212 """ 

213 

214 if ref.can_id is not None: 

215 base = ref.can_id & ~0xFF 

216 return [frame for frame in frames_on_bus.values() if frame.can_id == base] 

217 if ref.frame_ref is not None: 

218 frame = frames_on_bus.get(ref.frame_ref) 

219 return [frame] if frame is not None else [] 

220 return []

233def _validate_j1939_ref(owner: str, field_name: str, ref: J1939FrameRef, bus_frames: Dict[str, J1939Frame], bus_ref: str) -> None: 

234 """Resolve one J1939 sender / receiver reference against its bus, raising on any inconsistency.""" 

235 

236 if ref.frame_ref is not None and ref.can_id is not None: 

237 frame_by_name = bus_frames.get(ref.frame_ref) 

238 if frame_by_name is None: 

239 raise err_major( 

240 "{owner}: {field}: frame reference '{frame}' does not resolve to any J1939 frame declared on bus '{bus}' under {catalog}.", 

241 owner=owner, 

242 field=field_name, 

243 frame=ref.frame_ref, 

244 bus=bus_ref, 

245 catalog=_CAN_BUSES_CATALOG, 

246 category=Category.REFERENCE, 

247 error_number="372", 

248 ) 

249 # A frame name and a CAN identifier are consistent only if the named frame's derived 

250 # identifier (SA stripped) equals the reference's identifier (SA stripped). Several frames 

251 # on a bus may share the same derived identifier (same PGN / priority), so the id check 

252 # must be made against the *named* frame, never by picking an arbitrary frame with a 

253 # matching id and comparing names. 

254 base = ref.can_id & ~0xFF 

255 if frame_by_name.can_id != base: 

256 raise err_major( 

257 "{owner}: {field}: name '{frame}' and CAN identifier {can_id} do not agree " 

258 "— the frame named has derived identifier {derived} on bus '{bus}'.", 

259 owner=owner, 

260 field=field_name, 

261 frame=ref.frame_ref, 

262 can_id=f"0x{ref.can_id:x}", 

263 derived=f"0x{frame_by_name.can_id:x}", 

264 bus=bus_ref, 

265 category=Category.CONSISTENCY, 

266 error_number="423", 

267 ) 

268 return 

269 

270 matches = _resolve_j1939_frame_ref(ref, bus_frames) 

271 if not matches: 

272 raise err_major( 

273 "{owner}: {field}: frame reference {frame} does not resolve to any J1939 frame declared on bus '{bus}' under {catalog}.", 

274 owner=owner, 

275 field=field_name, 

276 frame=_render_j1939_ref(ref), 

277 bus=bus_ref, 

278 catalog=_CAN_BUSES_CATALOG, 

279 category=Category.REFERENCE, 

280 error_number="424", 

281 ) 

282 if len(matches) > 1: 

283 other = ", ".join(sorted(frame.name for frame in matches[1:])) 

284 raise err_major( 

285 "{owner}: {field}: CAN identifier {can_id} is ambiguous on bus '{bus}': it matches more than " 

286 "one J1939 frame ({first}, {other}). Name the frame with frame_ref to disambiguate.", 

287 owner=owner, 

288 field=field_name, 

289 can_id=f"0x{ref.can_id:x}", 

290 bus=bus_ref, 

291 first=matches[0].name, 

292 other=other, 

293 catalog=_CAN_BUSES_CATALOG, 

294 category=Category.CONSISTENCY, 

295 error_number="425", 

296 )

299def _validate_j1939_interface(iface: CANInterface, j1939_frames_by_bus: Dict[str, Dict[str, J1939Frame]]) -> None: 

300 """Resolve every J1939 sender / receiver frame reference on *iface* against its own bus. 

301 

302 ``J1939Frame`` may be addressed by name (a string), by its full on-bus CAN identifier (an integer), or by 

303 both. The catalog here is keyed by frame name. A ``can_id`` reference carries the emitter's Source Address 

304 in its low byte; the referenced frame is recovered by stripping those bits (``can_id & ~0xFF``) and 

305 matching the remainder against the frame's derived ``can_id``. A ``frame_ref`` reference is matched against 

306 the frame name directly. When both are given they must resolve to the *same* frame (double verification); 

307 a name/id mismatch is a consistency error. The bus is the interface's own ``bus_ref``, which is checked 

308 against the declared J1939 buses; a J1939 bus must exist for the node to attach to. 

309 """ 

310 

311 owner = f"CANInterface(name={iface.name})" 

312 

313 if iface.bus_ref not in j1939_frames_by_bus: 

314 raise err_major( 

315 "{owner}: bus_ref '{bus}' does not name any J1939 bus declared under {catalog}.", 

316 owner=owner, 

317 bus=iface.bus_ref, 

318 catalog=_CAN_BUSES_CATALOG, 

319 category=Category.REFERENCE, 

320 error_number="374", 

321 ) 

322 bus_frames = j1939_frames_by_bus[iface.bus_ref] 

323 for field_name in ("j1939_sender_frames", "j1939_receiver_frames"): 

324 for ref in getattr(iface, field_name) or []: 

325 _validate_j1939_ref(owner, field_name, ref, bus_frames, iface.bus_ref)

328def _emitter_sa_for_ref(ref: J1939FrameRef, iface: CANInterface, frames_on_bus: Dict[str, J1939Frame]) -> Tuple[str, int] | None: 

329 """Return ``(frame_name, source_address)`` an emitter *ref* would place on the wire, or ``None``. 

330 

331 A ``can_id`` emitter carries its own SA in the low byte; a ``frame_ref`` emitter uses the interface's 

332 ``source_address``. Either way the frame must resolve uniquely on the bus. 

333 """ 

334 

335 emitter: Tuple[str, int] | None = None 

336 if ref.can_id is not None: 

337 base = ref.can_id & ~0xFF 

338 matches = [f for f in frames_on_bus.values() if f.can_id == base] 

339 if len(matches) == 1: 

340 emitter = matches[0].name, ref.can_id & 0xFF 

341 # otherwise the can_id sender is ambiguous (384) or resolves to nothing (383), reported elsewhere 

342 elif ref.frame_ref is not None and iface.source_address is not None: 

343 frame = frames_on_bus.get(ref.frame_ref) 

344 if frame is not None: 

345 emitter = frame.name, iface.source_address 

346 return emitter

349def _build_j1939_emitter_sas(model: FLYNCModel, j1939_frames_by_bus: Dict[str, Dict[str, J1939Frame]]) -> Dict[Tuple[str, str], list[int]]: 

350 """Map ``(bus_ref, frame_name)`` to the source_addresses of every modelled J1939 emitter of that frame. 

351 

352 A reference that resolves to a frame by name needs a Source Address to form the full on-bus identifier. 

353 A sender supplies its own ``source_address``; a receiver observes a message sent by *another* node, so its 

354 SA must come from whoever is modelled as transmitting that frame on the bus. This pass collects, for every 

355 J1939 sender reference across the whole model, the SA each transmission would place on the wire and buckets 

356 them by ``(bus_ref, frame_name)`` so a receiver can look up the emitter's SA. 

357 

358 A sender reference contributes an SA when it carries an explicit ``can_id`` (whose SA byte is its own) or 

359 when it declares a ``source_address``. A sender that names a frame but declares no SA contributes nothing, 

360 so a receiver of that frame gets no SA and drops out of uniqueness (it is undecidable, not an error). 

361 """ 

362 

363 emitter_sas: Dict[Tuple[str, str], list[int]] = {} 

364 for controller, iface, kind in _iter_bus_interfaces(model): 

365 if not (isinstance(iface, CANInterface) and iface.is_j1939()): 

366 continue 

367 frames_on_bus = j1939_frames_by_bus.get(iface.bus_ref) 

368 if not frames_on_bus: 

369 continue # dangling bus_ref is reported elsewhere as 374 

370 for ref in iface.j1939_sender_frames or []: 

371 emitter = _emitter_sa_for_ref(ref, iface, frames_on_bus) 

372 if emitter is not None: 

373 emitter_sas.setdefault((iface.bus_ref, emitter[0]), []).append(emitter[1]) 

374 return emitter_sas

377def _full_id_for_frame(iface: CANInterface, field_name: str, frame: J1939Frame, emitter_sas: Dict[Tuple[str, str], list[int]]) -> int | None: 

378 """Return the full on-bus identifier *frame* gets under *field_name*, or ``None`` when it cannot be fixed. 

379 

380 A sender uses its own ``source_address``; a receiver has none of its own, so it borrows the SA of whoever 

381 is modelled as emitting the frame on the bus (from *emitter_sas*). No usable SA means ``None``. 

382 """ 

383 

384 if field_name == "j1939_sender_frames": 

385 # a sender without a declared SA is undecidable 

386 full_id = None if iface.source_address is None else frame.can_id | iface.source_address 

387 else: 

388 # a receiver borrows its emitter's SA; no known SA is undecidable 

389 sas = sorted(set(emitter_sas.get((iface.bus_ref, frame.name), ()))) 

390 full_id = frame.can_id | sas[0] if sas else None 

391 return full_id

394def _resolve_j1939_full_id( 

395 iface: CANInterface, 

396 field_name: str, 

397 ref: J1939FrameRef, 

398 frames_on_bus: Dict[str, J1939Frame], 

399 emitter_sas: Dict[Tuple[str, str], list[int]], 

400) -> Tuple[J1939Frame | None, int | None]: 

401 """Resolve *ref* and form its full on-bus J1939 identifier, or ``(None, None)`` when it cannot be fixed. 

402 

403 Returns ``(frame, full_id)``. ``full_id`` is the complete 29-bit identifier the reference places on (or 

404 observes on) the wire, Source Address included: 

405 

406 * a reference with an explicit ``can_id`` — sender or receiver — fixes the full id to ``ref.can_id``, whose 

407 SA byte is already set; 

408 * a sender named only by ``frame_ref`` derives the full id from the frame's own ``can_id`` OR the 

409 interface's ``source_address`` (the node transmits, so it uses its own SA); 

410 * a receiver named only by ``frame_ref`` has no SA of its own (it does not transmit): its SA is that of 

411 whoever is modelled as sending the frame on the bus, looked up in *emitter_sas*. When no emitter with a 

412 known SA exists, no full id can be fixed and ``(None, None)`` is returned — the reference is undecidable, 

413 not an error. 

414 

415 Returns ``(None, None)`` when the reference is unresolvable (reported elsewhere as 372/383/384) or when no 

416 full id can be determined. 

417 """ 

418 

419 frame: J1939Frame | None = None 

420 full: int | None = None 

421 if ref.can_id is not None: 

422 base = ref.can_id & ~0xFF 

423 matches = [f for f in frames_on_bus.values() if f.can_id == base] 

424 if len(matches) == 1: 

425 frame = matches[0] 

426 full = ref.can_id 

427 elif ref.frame_ref is not None: 

428 frame = frames_on_bus.get(ref.frame_ref) 

429 if frame is not None: 

430 full = _full_id_for_frame(iface, field_name, frame, emitter_sas) 

431 return (frame, full) if frame is not None and full is not None else (None, None)

434def _register_j1939_full_id( 

435 controller: Controller, 

436 iface: CANInterface, 

437 field_name: str, 

438 ref: J1939FrameRef, 

439 j1939_frames_by_bus: Dict[str, Dict[str, J1939Frame]], 

440 emitter_sas: Dict[Tuple[str, str], list[int]], 

441 bus_full_ids: Dict[str, Dict[int, Tuple[str, str]]], 

442) -> None: 

443 """Resolve *ref* and flag a collision when its full on-bus id was already used by a different frame.""" 

444 

445 owner = f"CANInterface(name={iface.name})" 

446 frames_on_bus = j1939_frames_by_bus.get(iface.bus_ref) 

447 if not frames_on_bus: 

448 return # a dangling bus_ref is already reported as 374 

449 frame, full = _resolve_j1939_full_id(iface, field_name, ref, frames_on_bus, emitter_sas) 

450 if frame is None or full is None: 

451 return # unresolvable or no fixed id, reported elsewhere 

452 seen = bus_full_ids.setdefault(iface.bus_ref, {}) 

453 prior = seen.get(full) 

454 if prior is not None and prior[0] != frame.name: 

455 err = err_major( 

456 "{owner}: {field} reference to frame '{frame}' yields CAN identifier {can_id} on bus " 

457 "'{bus}', already used by {prior}. A J1939 identifier is the frame's derived id OR'd " 

458 "with the node's source address and must be unique per bus.", 

459 owner=owner, 

460 field=field_name, 

461 frame=_render_j1939_ref(ref), 

462 can_id=hex(full), 

463 bus=iface.bus_ref, 

464 prior=prior[1], 

465 category=Category.UNIQUENESS, 

466 error_number="427", 

467 ) 

468 raise _with_source(err, _interface_locator(controller, iface, "can")) from None 

469 seen[full] = (frame.name, f"{owner}: {field_name} '{_render_j1939_ref(ref)}'")

472def _validate_j1939_unique_can_ids_on_bus(model: FLYNCModel, j1939_frames_by_bus: Dict[str, Dict[str, J1939Frame]]) -> None: 

473 """Require a unique full on-bus J1939 identifier per bus across all node references. 

474 

475 The identifier that actually appears on the wire is not a J1939 frame's bare ``can_id``: its low byte is 

476 the *source address* of the node transmitting (or observing) it. The bus-level ``054`` rule compares raw 

477 frames in isolation and so cannot see the SA; this workspace-level pass therefore gathers every J1939 

478 sender / receiver reference across all nodes sharing a bus, forms the full on-bus identifier for each 

479 (see :func:`_resolve_j1939_full_id`), and flags any full identifier that two references would put on the 

480 wire under *different* frames. 

481 

482 Two references that resolve to the *same* frame on the same bus are allowed to share a full id — that is 

483 the normal emitter + its receivers pattern, and the multi-node case where several nodes each emit the 

484 same Parameter Group (each with its own distinct SA, so distinct full ids). Only two *distinct* frames 

485 collapsing to the same on-bus id are a true collision. 

486 """ 

487 

488 bus_full_ids: Dict[str, Dict[int, Tuple[str, str]]] = {} 

489 emitter_sas = _build_j1939_emitter_sas(model, j1939_frames_by_bus) 

490 for controller, iface, kind in _iter_bus_interfaces(model): 

491 if not (isinstance(iface, CANInterface) and iface.is_j1939()): 

492 continue 

493 for field_name in ("j1939_sender_frames", "j1939_receiver_frames"): 

494 for ref in getattr(iface, field_name) or []: 

495 _register_j1939_full_id(controller, iface, field_name, ref, j1939_frames_by_bus, emitter_sas, bus_full_ids)

498def validate_interface_frame_refs(model: FLYNCModel) -> None: 

499 """Workspace pass: every CAN / LIN interface must name a declared bus of its own kind and resolve its frame refs. 

500 

501 The check is a plain lookup - is this frame id declared on this bus - performed against the catalog of the 

502 interface's own bus kind. Both aspects are load-bearing: 

503 

504 *Scoped by bus*, because a CAN id is only unique within a bus: id ``0x100`` on ``CAN1`` and on ``CAN2`` are 

505 two different frames. The bus is the interface's own ``bus_ref``; a 

506 :class:`~flync.model.flync_4_ecu.can_interface.CANFrameRef` carries the ``frame_ref`` plus its ``id_format`` 

507 (a standard and an extended frame may share a numeric id). Asking only whether an id exists *somewhere* 

508 would accept an interface on ``CAN1`` declaring a frame that only exists on ``CAN2``. 

509 

510 *Scoped by bus kind*, because ``bus_ref`` is a plain string and ``frame_ref`` a plain int - the type system 

511 stops a ``LINFrameRef`` from landing in a ``CANInterface``, but nothing stops a CAN interface from naming a 

512 LIN bus. Resolving a CAN interface against ``can_buses`` only is what catches that, so no separate cross-kind 

513 rule is needed: the LIN bus name simply is not in the CAN catalog. A single catalog merged over both 

514 kinds would instead accept LIN ids on a CAN interface, and would be ambiguous anyway - bus names are unique 

515 per kind but not across kinds, so a workspace may hold a CAN bus and a LIN bus of the same name. 

516 

517 A bus kind whose catalog is empty is skipped: FLYNC supports partial models (an ECU or controller may be 

518 modeled without the bus catalog it will later be wired into), and there is nothing to resolve against. 

519 Once a workspace declares buses of a kind, every interface of that kind must resolve. 

520 """ 

521 

522 can_frames_by_bus = _build_can_frames_by_bus(model) 

523 lin_frames_by_bus = _build_lin_frames_by_bus(model) 

524 j1939_frames_by_bus = _build_j1939_frames_by_bus(model) 

525 

526 for controller, iface, kind in _iter_bus_interfaces(model): 

527 frames_by_bus: Dict[str, Dict[FrameKey, AnyFrame]] = can_frames_by_bus if kind == "can" else lin_frames_by_bus # type: ignore[assignment] 

528 if not frames_by_bus: 

529 continue 

530 try: 

531 _validate_interface(iface, kind, frames_by_bus) 

532 except PydanticCustomError as err: 

533 raise _with_source(err, _interface_locator(controller, iface, kind)) from None 

534 if isinstance(iface, CANInterface) and iface.is_j1939(): 

535 try: 

536 _validate_j1939_interface(iface, j1939_frames_by_bus) 

537 except PydanticCustomError as err: 

538 raise _with_source(err, _interface_locator(controller, iface, "can")) from None
