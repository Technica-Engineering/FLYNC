103def validate_cbs_idleslopes_fit_portspeed(traffic_classes: list[TrafficClass] | None, port_speed: int | None) -> list[TrafficClass] | None: 

104 """ 

105 Custom Validator for a list of Traffic Classes to check conformity to MII/MDI speed. 

106 

107 Args: 

108 traffic_classes (list): List of element type `TrafficClass`. 

109 

110 port_speed (int): MII or MDI speed of the port. 

111 

112 Raises: 

113 err_major: The sum of idleslopes of all shapers on one port must be equal or lower than the port speed. 

114 

115 Returns: 

116 list: Return list of traffic classes as received. 

117 """ 

118 

119 if not traffic_classes: 

120 return None 

121 if not port_speed: 

122 raise err_major( 

123 "Cannot validate Traffic Classes! No port speed defined. Make sure to configure MII or MDI.", 

124 category=Category.REQUIRED, 

125 error_number="010",

129 

130 for tr_class in traffic_classes:

137 category=Category.CONSISTENCY, 

138 error_number="011",

144 """

462def validate_mii_subports(ports: Sequence[MIIPort], label: str, uplink_ports: frozenset[str] = frozenset()) -> None: 

463 """ 

464 Validate ``mii_subport_of``/``mii_subport_id`` references between ports sharing a multi-port xMII link 

465 (see :class:`~flync.model.flync_4_ecu.mii.USXGMII`/:class:`~flync.model.flync_4_ecu.mii.USGMII`). 

466 

467 Args: 

468 ports (list): Ports to check, each exposing ``name``, ``mii_config``, ``mii_subport_of``, ``mii_subport_id``. 

469 

470 label (str): Human-readable label for the owning port list (used in error messages). 

471 

472 uplink_ports (frozenset[str]): Names of ports that terminate the channelized uplink of a port extender. Such a port is a 

473 pure lane: it carries the multi-port link but occupies none of its channels, which the extender ports claim instead. 

474 

475 Raises: 

476 err_major: A port sets both ``mii_config`` and ``mii_subport_of``. 

477 

478 err_major: ``mii_subport_of`` does not name another port in ``ports``. 

479 

480 err_major: The referenced port's ``mii_config`` is not a multi-port USXGMII/USGMII link. 

481 

482 err_major: ``mii_subport_id`` is not one of the referenced link's channel ``subport_id`` values. 

483 

484 err_major: A port carrying a multi-port link has a ``mii_subport_id`` (default 0) that is not one of its own channels. 

485 

486 err_major: Two ports claim the same channel of the same link. The port carrying the link claims the channel 

487 named by its own ``mii_subport_id`` (default 0); subports claim the others. 

488 

489 err_major: A listed channel is claimed by no port. 

490 

491 err_major: A port's ``mdi_config`` speed (the carrying port's included) is one its channel cannot carry (the rule of 

492 :meth:`~flync.model.flync_4_ecu.port.ECUPort.verify_mdi_and_mii_config_speeds_are_compatible`, applied per channel). 

493 Skipped for a port that has no ``mdi_config`` of its own yet, such as a switch port. 

494 """ 

495 

496 by_name = {port.name: port for port in ports} 

497 claimed: list[tuple[str, int]] = [] 

498 subports: list[tuple[MIIPort, MIIPort]] = [] 

499 

500 for port in ports: 

501 if port.mii_subport_of is None: 

502 continue 

503 sibling = _resolve_subport_link(port, port.mii_subport_of, by_name, label) 

504 claimed.append((port.mii_subport_of, port.mii_subport_id or 0)) 

505 subports.append((port, sibling)) 

506 

507 # The port carrying a link occupies one of its channels too: the one named by its own mii_subport_id, default 0. 

508 for port in ports: 

509 if not isinstance(port.mii_config, (USXGMII, USGMII)) or port.name in uplink_ports: 

510 continue 

511 own_channel = _own_channel(port, port.mii_config, label) 

512 claimed.append((port.name, own_channel)) 

513 subports.append((port, port)) 

514 

515 validate_list_items_unique(claimed, f"{label} mii_subport_of/mii_subport_id") 

516 

517 claimed_set = set(claimed) 

518 for port in ports: 

519 if isinstance(port.mii_config, (USXGMII, USGMII)): 

520 _validate_channels_claimed(port, port.mii_config, claimed_set, label) 

521 

522 for port, link in subports: 

523 mdi_config = getattr(port, "mdi_config", None) 

524 if mdi_config is not None: 

525 validate_mdi_speed_fits_channel(label, port, link, mdi_config)

528def _resolve_subport_link(port: MIIPort, subport_of: str, by_name: dict[str, MIIPort], label: str) -> MIIPort: 

529 """ 

530 Resolve and validate the multi-port link a subport refers to via ``mii_subport_of``. 

531 

532 Args: 

533 port (object): The port that sets ``mii_subport_of``. 

534 

535 subport_of (str): The value of that ``mii_subport_of``. 

536 

537 by_name (dict): Ports of the owning list, keyed by name. 

538 

539 label (str): Human-readable label for the owning port list (used in error messages). 

540 

541 Returns: 

542 object: The port carrying the referenced multi-port link. 

543 

544 Raises: 

545 err_major: ``port`` also sets ``mii_config``, the reference is not another port, the referenced port carries no 

546 multi-port link, or ``mii_subport_id`` is not one of the link's channels. 

547 """ 

548 

549 if port.mii_config is not None: 

550 raise err_major( 

551 "{label} '{port}' sets both mii_config and mii_subport_of; a channel of a shared multi-port link does not have its own mii_config.", 

552 label=label, 

553 port=port.name, 

554 category=Category.CONSISTENCY, 

555 error_number="383", 

556 ) 

557 

558 sibling = by_name.get(subport_of) 

559 if sibling is None or sibling.name == port.name: 

560 raise err_major( 

561 "{label} '{port}' has mii_subport_of '{subport_of}', which is not another port in {label}.", 

562 label=label, 

563 port=port.name, 

564 subport_of=port.mii_subport_of, 

565 category=Category.REFERENCE, 

566 error_number="384", 

567 ) 

568 

569 if not isinstance(sibling.mii_config, (USXGMII, USGMII)): 

570 raise err_major( 

571 "{label} '{port}' has mii_subport_of '{sibling}', which does not carry a multi-port USXGMII/USGMII link.", 

572 label=label, 

573 port=port.name, 

574 sibling=sibling.name, 

575 category=Category.COMPATIBILITY, 

576 error_number="385", 

577 ) 

578 

579 channel_subport_ids = {channel.subport_id for channel in sibling.mii_config.channels} 

580 if port.mii_subport_id not in channel_subport_ids: 

581 raise err_major(

595 """

658def validate_mdi_speed_fits_channel(label: str, port: MIIPort, link: MIIPort, mdi_config: MDIConfig) -> None: 

659 """ 

660 Check that the channel a port occupies on a multi-port xMII link can carry the port's MDI speed. 

661 

662 This is the rule of :meth:`~flync.model.flync_4_ecu.port.ECUPort.verify_mdi_and_mii_config_speeds_are_compatible`, applied to 

663 one channel of the link. 

664 

665 Args: 

666 label (str): Human-readable label for the owning port list (used in error messages). 

667 

668 port (object): The port whose channel is checked; its ``mii_subport_id`` (default 0) picks the channel. 

669 

670 link (object): The port whose ``mii_config`` is the multi-port link (``port`` itself when it carries the link). 

671 

672 mdi_config (object): MDI configuration whose ``speed`` is compared against the channel. 

673 

674 Raises: 

675 err_major: The channel's speed is lower than the MDI speed. 

676 """ 

677 

678 lane = link.mii_config 

679 if not isinstance(lane, (USXGMII, USGMII)): 

680 return 

681 channel = next(c for c in lane.channels if c.subport_id == (port.mii_subport_id or 0)) 

682 if not mii_carries_mdi_speed(lane, mdi_config.speed, channel.speed): 

683 raise err_major( 

684 "{label} '{port}' has an MDI speed of {mdi_speed} Mbps, which channel {subport_id} of '{link}' ({channel_speed} Mbps) cannot carry.", 

685 label=label, 

686 port=port.name, 

687 mdi_speed=mdi_config.speed, 

688 subport_id=channel.subport_id, 

689 link=link.name, 

690 channel_speed=channel.speed, 

691 category=Category.CONSISTENCY, 

692 error_number="388", 

693 )

696def validate_switch_port_mdi_speed(switch_port: SwitchPort) -> None: 

697 """ 

698 Check a switch port's MDI speed against the xMII it sits on, once the MDI config has been copied in. 

699 

700 ``SwitchPort.mdi_config`` is only set when the internal topology binds a connection, so 

701 :func:`validate_mii_subports` cannot see it; this runs after that step. It covers a port that carries a multi-port link 

702 (its own channel), a subport (the channel it claims on the sibling's link) and a port with a plain single-port xMII. 

703 

704 Args: 

705 switch_port (object): The switch port, already holding its ``mdi_config``. Nothing is checked while that is ``None``. 

706 

707 Raises: 

708 err_major: The xMII, or the channel the port occupies, cannot carry the MDI speed. 

709 """ 

710 

711 mdi_config = switch_port.mdi_config 

712 if mdi_config is None: 

713 return 

714 

715 label = "Switch Ports" 

716 link = None 

717 if isinstance(switch_port.mii_config, (USXGMII, USGMII)): 

718 link = switch_port 

719 elif switch_port.mii_subport_of is not None: 

720 link = next((p for p in switch_port.get_switch().ports if p.name == switch_port.mii_subport_of), None) 

721 

722 if link is not None and isinstance(link.mii_config, (USXGMII, USGMII)): 

723 validate_mdi_speed_fits_channel(label, switch_port, link, mdi_config) 

724 return 

725 

726 mii_config = switch_port.mii_config 

727 if mii_config is not None and mii_config.speed is not None and not mii_carries_mdi_speed(mii_config, mdi_config.speed): 

728 raise err_major( 

729 "{label} '{port}' has an MDI speed of {mdi_speed} Mbps, which its MII config ({mii_speed} Mbps) cannot carry.", 

730 label=label, 

731 port=switch_port.name, 

732 mdi_speed=mdi_config.speed, 

733 mii_speed=mii_config.speed, 

734 category=Category.CONSISTENCY, 

735 error_number="390", 

736 )
