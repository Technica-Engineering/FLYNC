1""" 

2Base contract and registry for FLYNC workspace schema migrations. 

3 

4A migration transforms a workspace's YAML *data* (never Python objects) from one FLYNC schema 

5to a newer one, so that workspaces exchanged between suppliers and OEMs can be brought up to the 

6installed release without losing content. Migrations operate on the raw parsed documents before 

7any Pydantic validation runs, which is what allows re-keying, restructure and cross-document moves 

8(the workspace is treated as data, consistent with ``root_model`` never being persisted). 

9 

10The chain is keyed by the FLYNC PEP 440 release: each migration declares the ``target_version`` 

11that introduced its change. A workspace recorded at version ``R`` is migrated to the installed 

12version ``T`` by applying, in ascending ``target_version`` order, every migration ``m`` for which 

13``R < m.target_version <= T``. 

14""" 

15

23#: Documents surface handed to migrations: workspace-relative URI → parsed YAML mapping/list. 

24#: This mirrors the ``ast`` of each :class:`~flync.sdk.workspace.document.Document`. 

25Documents = Dict[str, Any] 

26 

27#: Oldest recorded FLYNC release for which a migration path to the current release is guaranteed. 

28#: 

29#: The migration history in ``docs/source/model_change_history.rst`` begins at 0.9.x, so any 

30#: workspace older than this cannot be migrated and must be re-created / re-imported. This value 

31#: must be raised (not lowered) as migrations are added; it can never move backwards. 

32LOWEST_SUPPORTED_VERSION = "0.9.0"

35class FLYNCMigration(ABC): 

36 """ 

37 A single schema migration from an older FLYNC release to a newer one. 

38 

39 Subclass this, set the class attributes, and implement :meth:`apply` (and optionally 

40 :meth:`should_apply`). Decorate the subclass with :func:`migration` (or call 

41 :meth:`MigrationRegistry.register`) so the mechanism discovers it. The base class is abstract 

42 and cannot be instantiated directly. 

43 

44 Attributes: 

45 unique_id (str): Stable, globally-unique identifier for this migration (should never 

46 change once shipped - it is written to the workspace history manifest). 

47 

48 target_version (str): PEP 440 FLYNC release that introduced this schema change. A 

49 workspace older than this needs the migration applied to reach the new format. 

50 

51 description (str): Human-readable summary of the change, shown by ``flync migrate``. 

52 

53 error_ids (tuple[str, ...]): ``FLYNC-...`` error IDs from ``model_change_history.rst`` 

54 that this migration resolves. 

55 """ 

56 

57 unique_id: str = "" 

58 target_version: str = "" 

59 description: str = "" 

60 error_ids: tuple[str, ...] = () 

61 

62 def should_apply(self, documents: Documents) -> bool: 

63 """ 

64 Whether this particular workspace actually needs this transform. 

65 

66 Runs before :meth:`apply` (once the workspace is already known to be older than 

67 ``target_version``) so migrations can be a no-op for workspaces that happen to already 

68 conform. Defaults to ``True``; override when the trigger can be detected precisely. 

69 

70 Args: 

71 documents: The parsed workspace documents, keyed by URI. 

72 

73 Returns: 

74 bool: ``True`` to record the migration as applied to this workspace. 

75 """ 

76 return True 

77 

78 @abstractmethod 

79 def apply(self, documents: Documents) -> None: 

80 """ 

81 Transform ``documents`` in place from the previous schema to ``target_version``. 

82 

83 Must be idempotent: running it on documents it already migrated has to leave them 

84 unchanged. An interrupted migration rewrites some documents before the version stamp is 

85 advanced, and the next run then applies the whole chain again. 

86 

87 Args: 

88 documents: The parsed workspace documents, keyed by URI. Mutate these; do not 

89 return a new mapping. Only URIs whose content actually changes are rewritten 

90 to disk - untouched documents keep their original formatting. 

91 """ 

92 raise NotImplementedError(f"{type(self).__name__}.apply is not implemented") 

93 

94 def __repr__(self) -> str: 

95 return f"<{type(self).__name__} {self.unique_id} → {self.target_version}>"

98class MigrationRegistry: 

99 """ 

100 Holds every :class:`FLYNCMigration` and validates their invariants. 

101 

102 This is a **singleton**: ``MigrationRegistry()`` always returns the one process-wide instance 

103 (also exposed as :data:`migration_registry` and populated by importing 

104 :mod:`flync.core.version_migrators.migrations`), so migrations registered from anywhere are 

105 visible to the engine and CLI. 

106 

107 Tests (or other code needing an isolated, disposable set of migrations) create an independent 

108 registry with :meth:`new` and pass it explicitly to the engine. 

109 """ 

110 

111 _singleton: MigrationRegistry | None = None 

112 

113 def __new__(cls) -> MigrationRegistry: 

114 if cls._singleton is None: 

115 cls._singleton = super().__new__(cls) 

116 return cls._singleton 

117 

118 @classmethod 

119 def new(cls) -> MigrationRegistry: 

120 """Return a fresh, independent registry (for tests / bespoke migration sets).""" 

121 obj = super().__new__(cls) 

122 obj._migrations = {} 

123 return obj 

124 

125 def __init__(self) -> None: 

126 # Only initialize the backing store once; `new()` pre-seeds it and the singleton must not 

127 # be reset by a subsequent `MigrationRegistry()` call. 

128 if not hasattr(self, "_migrations"): 

129 self._migrations: dict[str, FLYNCMigration] = {} 

130 

131 def register(self, item: FLYNCMigration | type[FLYNCMigration]) -> FLYNCMigration: 

132 """ 

133 Register a migration (instance or class), enforcing uniqueness of ids and targets. 

134 

135 Args: 

136 item: A migration class to instantiate, or an already-instantiated migration. 

137 

138 Returns: 

139 FLYNCMigration: The registered (instantiated) migration. 

140 

141 Raises: 

142 ValueError: If ``unique_id`` is empty/duplicated, ``target_version`` is not a PEP 440 

143 version, or two migrations share a ``target_version``. 

144 """ 

145 migration = item() if isinstance(item, type) else item 

146 if not migration.unique_id: 

147 raise ValueError("A migration must declare a non-empty 'unique_id'.") 

148 try: 

149 Version(migration.target_version) 

150 except InvalidVersion as exc: 

151 raise ValueError( 

152 f"Migration {migration.unique_id!r} has target_version {migration.target_version!r}, which is not a PEP 440 version." 

153 ) from exc 

154 if migration.unique_id in self._migrations: 

155 raise ValueError(f"Duplicate migration unique_id: {migration.unique_id!r}.") 

156 for existing in self._migrations.values(): 

157 if existing.target_version == migration.target_version: 

158 raise ValueError( 

159 f"Migrations {existing.unique_id!r} and {migration.unique_id!r} share target_version " 

160 f"{migration.target_version!r}; target versions must be unique to keep the chain ordered." 

161 ) 

162 self._migrations[migration.unique_id] = migration 

163 return migration 

164 

165 def get(self, unique_id: str) -> FLYNCMigration | None: 

166 """Return the migration with ``unique_id``, or ``None``.""" 

167 return self._migrations.get(unique_id) 

168 

169 def all(self) -> list[FLYNCMigration]: 

170 """All registered migrations.""" 

171 return list(self._migrations.values()) 

172 

173 def clear(self) -> None: 

174 """Remove all registrations (used to reset the default registry under test).""" 

175 self._migrations.clear()

183def migration(cls: type[FLYNCMigration]) -> type[FLYNCMigration]: 

184 """ 

185 Decorator that registers a :class:`FLYNCMigration` subclass into :data:`migration_registry`. 

186 

187 Use on every concrete migration class defined under 

188 :mod:`flync.core.version_migrators.migrations`: 

189 

190 .. code-block:: python 

191 

192 @migration 

193 class MyMigration(FLYNCMigration): 

194 unique_id = "my_rename" 

195 target_version = "0.15.0" 

196 description = "Renames the 'foo' key to 'bar'." 

197 

198 def apply(self, documents): 

199 for uri, content in documents.items(): 

200 if isinstance(content, dict) and "foo" in content: 

201 content["bar"] = content.pop("foo") 

202 """ 

203 migration_registry.register(cls()) 

204 return cls
