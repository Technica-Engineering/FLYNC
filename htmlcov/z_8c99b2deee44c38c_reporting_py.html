1"""Per-conversion reporting. 

2 

3A conversion has any number of inputs and outputs. Each of them is an endpoint 

4with its own reports folder, by default ``<endpoint>/.flync/reports`` (see 

5:attr:`~flync_converter.ConverterConfig.report_dir`), holding one folder for 

6its converter:: 

7 

8 <reports>/<converter_name>/config.yaml configuration the converter ran with 

9 <reports>/<converter_name>/logs.txt that converter's records 

10 <reports>/<converter_name>/report.yaml what the converter reported (its reporters' files) 

11 <reports>/<converter_name>/... files the converter writes itself 

12 

13The reports folder of every output also holds the report of the conversion as 

14a whole:: 

15 

16 <reports>/logs.txt every captured record 

17 <reports>/report.yaml inputs, outputs, merge conflicts, model counts, status (its reporters' files) 

18 

19An endpoint that is a single file uses the folder the file is in. An endpoint 

20with ``report_enabled`` set to ``False``, or whose reports folder cannot be 

21written, gets no report; the conversion runs all the same. A source that does 

22not exist is not created for its report. 

23 

24The logs use the standard :mod:`logging` hierarchy. The base library defines 

25the main ``flync_converter`` logger. The conversion log captures it, so it holds 

26every converter sub-logger (e.g. ``flync_converter.converters.flync_converter``), 

27plus every logger the converters list in 

28:attr:`~flync_converter.base.BaseConverter.report_loggers`. A converter's own 

29log captures only the loggers that converter lists, and only while that 

30converter runs when the conversion marks it with :meth:`ConversionReport.step`. 

31 

32No other logger is captured or has its level changed. Loggers exist once per 

33process, so one conversion at a time is supported per process: a second 

34conversion running concurrently also writes its records into the first one's 

35report. 

36"""

76def _outermost(names: Iterable[str]) -> list[str]: 

77 """Drop every logger name that has an ancestor in ``names``. 

78 

79 A record propagates through all its ancestors, so a handler attached to 

80 both ``flync`` and ``flync.sdk`` would write each ``flync.sdk`` record twice. 

81 

82 Args: 

83 names: Dotted logger names. 

84 

85 Returns: 

86 The names without a listed ancestor, sorted and without duplicates. 

87 """ 

88 unique = sorted(set(names)) 

89 return [name for name in unique if not any(name.startswith(f"{other}.") for other in unique)]

186def _endpoint(converter: BaseConverter, is_input: bool) -> _Endpoint: 

187 """Describe a converter taking part in a conversion, from its configuration.""" 

188 config = converter.config 

189 if not isinstance(config, ConverterConfig): 

190 return _Endpoint(converter, None, logging.INFO, {"converter": converter.name, "path": None}) 

191 root = config_reports_root(config) if config.report_enabled else None 

192 # A missing source is not created for the sake of its report. 

193 if is_input and config.report_dir is None and not Path(config.config_path).exists(): 

194 root = None 

195 return _Endpoint(converter, root, config.report_level, {"converter": converter.name, "path": config.config_path})

198class ConversionReport: 

199 """Context manager writing the report of one conversion. 

200 

201 Every converter passed in must already hold the configuration it runs 

202 with: its ``report_enabled``, ``report_min_log_level`` and ``report_dir`` 

203 decide whether, how and where its part of the report is written. A 

204 converter without a configuration gets no report. 

205 

206 On entry: 

207 

208 * the conversion log ``logs.txt`` is created in the reports folder of every 

209 output, with a :class:`logging.FileHandler` attached to the 

210 ``flync_converter`` logger and to the ``report_loggers`` of all 

211 converters; 

212 * each converter gets its folder ``<reports>/<name>/``, exposed to it as 

213 :attr:`~flync_converter.base.BaseConverter.report_dir`, holding 

214 ``config.yaml`` (every value of the configuration the converter runs 

215 with, defaults included) and, when it lists ``report_loggers``, its own 

216 ``logs.txt`` capturing only those loggers; 

217 * each converter gets an active 

218 :class:`~flync_converter.base.ConverterReport` as 

219 :attr:`~flync_converter.base.BaseConverter.report`; 

220 * every captured logger has its level lowered to the lowest level a log 

221 asks for when it would otherwise drop those records. 

222 

223 A reports folder that cannot be created is logged as a warning and left 

224 out. 

225 

226 On exit, while the logs are still open: 

227 

228 * when the block raised, the exception and its traceback are written to the 

229 conversion log as an ``ERROR`` record. The record goes to the conversion 

230 log only, not to other handlers, and the exception propagates unchanged; 

231 * the conversion's ``report.yaml`` is written into the reports folder of 

232 every output: the inputs and outputs with their report folders, the 

233 merge record when :meth:`record_merge` was called, the model counts when 

234 :meth:`record_model` was called, and the status (``succeeded``, or 

235 ``failed`` with the error); 

236 * each converter's report is written by its 

237 :attr:`~flync_converter.base.BaseConverter.reporters`. A reporter that 

238 fails is logged and skipped. 

239 

240 Then the log handlers are detached and closed, every logger's previous 

241 level is restored, and each converter's ``report_dir`` and ``report`` are 

242 reset. Every file replaces the one of a previous conversion. 

243 

244 Example: 

245 >>> with ConversionReport([source_converter], [destination_converter]) as report: 

246 ... with report.step(source_converter): 

247 ... model = source_converter.decode() 

248 ... report.record_model(model) 

249 ... with report.step(destination_converter): 

250 ... destination_converter.encode(model) 

251 """

253 def __init__( 

254 self, 

255 inputs: Sequence[BaseConverter], 

256 outputs: Sequence[BaseConverter], 

257 reporters: Sequence[BaseReporter] = DEFAULT_REPORTERS, 

258 ) -> None: 

259 """Prepare the report without touching the file system or logging. 

260 

261 Args: 

262 inputs: The converters reading the sources, configured. 

263 outputs: The converters writing the destinations, configured. 

264 reporters: Reporters writing the conversion's report into the 

265 reports folder of every output, one file each. 

266 

267 Raises: 

268 ValueError: If two converters of one name share a reports folder. 

269 """ 

270 self.reporters = tuple(reporters) 

271 self._inputs = [_endpoint(converter, is_input=True) for converter in inputs] 

272 self._outputs = [_endpoint(converter, is_input=False) for converter in outputs] 

273 self._endpoints = [*self._inputs, *self._outputs] 

274 self._reject_shared_folders() 

275 # Reports folder of every output and the lowest level any output sharing it asks for. 

276 self._roots: dict[Path, int] = {} 

277 for endpoint in self._outputs: 

278 if endpoint.root is not None: 

279 self._roots[endpoint.root] = min(endpoint.level, self._roots.get(endpoint.root, endpoint.level)) 

280 self._shared: dict[str, Any] = {} 

281 self._failed_outputs = False 

282 self._current: BaseConverter | None = None 

283 self._shared_handlers: list[logging.FileHandler] = [] 

284 self._handlers: list[tuple[logging.Handler, list[logging.Logger]]] = [] 

285 self._previous_levels: dict[logging.Logger, int] = {}

287 def _reject_shared_folders(self) -> None: 

288 """Refuse two converters writing their report into the same folder.""" 

289 seen: set[Path] = set() 

290 for endpoint in self._endpoints: 

291 folder = endpoint.folder 

292 if folder is None: 

293 continue 

294 resolved = folder.resolve() 

295 if resolved in seen: 

296 raise ValueError( 

297 f"Two '{endpoint.converter.name}' converters would write their report to {folder}. Give one of them its own report_dir." 

298 ) 

299 seen.add(resolved)

301 @property 

302 def roots(self) -> tuple[Path, ...]: 

303 """The reports folders holding the report of the conversion as a whole.""" 

304 return tuple(self._roots)

336 def step(self, converter: BaseConverter) -> Iterator[None]: 

337 """Mark ``converter`` as the one running, so only its own log captures its loggers. 

338 

339 Without it, two converters listing the same logger both capture every 

340 record of that logger. 

341 

342 Args:

345 previous, self._current = self._current, converter 

346 try: 

347 yield 

348 finally: 

349 self._current = previous

351 def __enter__(self) -> Self: 

352 """Create the report files and folders and start capturing records.""" 

353 all_converter_loggers = [name for endpoint in self._endpoints for name in endpoint.converter.report_loggers] 

354 unwritable = [] 

355 for root, level in self._roots.items(): 

356 try: 

357 root.mkdir(parents=True, exist_ok=True) 

358 self._shared_handlers.append(self._attach(root / LOG_FILENAME, [MAIN_LOGGER, *all_converter_loggers], level)) 

359 except OSError as error: 

360 logger.warning("Conversion report skipped, cannot write to %s: %s", root, error) 

361 unwritable.append(root) 

362 for root in unwritable: 

363 del self._roots[root] 

364 

365 for endpoint in self._endpoints: 

366 folder = endpoint.folder 

367 if folder is None: 

368 continue 

369 converter = endpoint.converter 

370 try: 

371 folder.mkdir(parents=True, exist_ok=True) 

372 if isinstance(converter.config, ConverterConfig): 

373 _write_config_record(folder / CONFIG_RECORD_FILENAME, converter.config) 

374 if converter.report_loggers: 

375 handler = self._attach(folder / LOG_FILENAME, converter.report_loggers, endpoint.level) 

376 handler.addFilter(self._while_running(converter)) 

377 except OSError as error: 

378 logger.warning("Report of converter %s skipped, cannot write to %s: %s", converter.name, folder, error) 

379 endpoint.root = None 

380 continue 

381 endpoint.entry["report"] = str(folder) 

382 converter.report_dir = folder 

383 converter.report = ConverterReport() 

384 return self

395 def _attach(self, path: Path, logger_names: Iterable[str], level: int) -> logging.FileHandler: 

396 """Open a log file and attach its handler to the given loggers. 

397 

398 Args: 

399 path: Log file to create, replacing an existing one. 

400 logger_names: Names of the loggers the file captures. 

401 level: Lowest severity the file captures. 

402 

403 Returns: 

404 The handler writing the file. 

405 """ 

406 handler = logging.FileHandler(path, mode="w", encoding="utf-8") 

407 handler.setLevel(level) 

408 handler.setFormatter(logging.Formatter(LOG_FORMAT)) 

409 loggers = [logging.getLogger(name) for name in _outermost(logger_names)] 

410 for log in loggers: 

411 self._previous_levels.setdefault(log, log.level) 

412 # A logger at NOTSET takes its effective level from its ancestors, which may drop 

413 # records below WARNING, so it is set explicitly. An existing more permissive level is kept. 

414 if log.level == logging.NOTSET or log.level > level: 

415 log.setLevel(level) 

416 log.addHandler(handler) 

417 self._handlers.append((handler, loggers)) 

418 return handler

420 def __exit__( 

421 self, 

422 exc_type: type[BaseException] | None, 

423 exc: BaseException | None, 

424 traceback: TracebackType | None, 

425 ) -> None: 

426 """Record the outcome, write the reports, then close the logs and restore the loggers.""" 

427 if exc is not None: 

428 record = logging.getLogger(MAIN_LOGGER).makeRecord( 

429 MAIN_LOGGER, logging.ERROR, __file__, 0, "Conversion failed: %s", (exc,), (type(exc), exc, traceback) 

430 ) 

431 for shared_handler in self._shared_handlers: 

432 shared_handler.handle(record) 

433 self._write_reports(exc) 

434 for handler, loggers in self._handlers: 

435 for log in loggers: 

436 log.removeHandler(handler) 

437 handler.close() 

438 for log, level in self._previous_levels.items(): 

439 log.setLevel(level) 

440 self._handlers.clear() 

441 self._shared_handlers.clear() 

442 self._previous_levels.clear() 

443 for endpoint in self._endpoints: 

444 endpoint.converter.report_dir = None 

445 endpoint.converter.report = INACTIVE_REPORT

447 def _write_reports(self, exc: BaseException | None) -> None: 

448 """Write the conversion report and each converter's report while the logs are still open.""" 

449 shared: dict[str, Any] = { 

450 "inputs": [endpoint.entry for endpoint in self._inputs], 

451 "outputs": [endpoint.entry for endpoint in self._outputs], 

452 **self._shared, 

453 "status": "failed" if exc is not None or self._failed_outputs else "succeeded", 

454 } 

455 if exc is not None: 

456 shared["error"] = f"{type(exc).__name__}: {exc}" 

457 for root in self._roots: 

458 write_with(self.reporters, shared, root) 

459 for endpoint in self._endpoints: 

460 if endpoint.folder is not None: 

461 endpoint.converter.report.write(endpoint.converter.reporters, endpoint.folder)
