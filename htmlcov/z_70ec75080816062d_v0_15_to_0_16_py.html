1""" 

2Migration of a FLYNC 0.15 workspace to the 0.16 schema. 

3 

4The registry allows one migration per release, so every breaking 0.16 change is a step of 

5:class:`Migration016`. Each step is idempotent on its own and resolves references with the lookup 

6rules 0.15 itself used, so a reference that loaded on 0.15 keeps pointing at the same frame. 

7See ``docs/source/model_change_history/v0_15_0_16.rst``. 

8""" 

9

13from typing import Any, Iterator

15from flync.core.version_migrators._base import Documents, FLYNCMigration, migration

17 

18logger = logging.getLogger(__name__)

20_PDU2_MIN_FORMAT = 240

22 

23def _pgn(frame: Node) -> int | None: 

24 """PGN of a J1939 frame document, as ``J1939Frame.pgn`` computes it; ``None`` when a field is missing.""" 

25 edp, dp, pf, ps = (frame.get(key) for key in ("extended_data_page", "data_page", "pdu_format", "pdu_specific")) 

26 if not (_is_can_id(edp) and _is_can_id(dp) and _is_can_id(pf) and _is_can_id(ps)): 

27 return None 

28 return (edp << 17) | (dp << 16) | (pf << 8) | (ps if pf >= _PDU2_MIN_FORMAT else 0) 

29 

30 

31def _j1939_frames(documents: Documents) -> Iterator[tuple[str, int, str]]: 

32 """Yield ``(bus name, pgn, frame name)`` for every named J1939 frame of a bus document.""" 

33 for content in documents.values(): 

34 for bus in (node for node in _walk(content) if isinstance(node.get("name"), str)): 

35 for frame in _dicts(bus.get("frames")): 

36 pgn = _pgn(frame) if frame.get("type") == "j1939" else None 

37 if pgn is not None and isinstance(frame.get("name"), str): 

38 yield bus["name"], pgn, frame["name"]

41def _j1939_frame_names_by_bus(documents: Documents) -> dict[str, dict[int, str | None]]: 

42 """ 

43 Return ``{bus name: {pgn: frame name}}``. 

44 

45 0.15 rejected two J1939 frames with one PGN on a bus (error 346), so a PGN names one frame. A PGN 

46 that several frames share maps to ``None``: the reference it is in cannot be resolved. 

47 """ 

48 buses: dict[str, dict[int, str | None]] = {} 

49 for bus, pgn, name in _j1939_frames(documents): 

50 names = buses.setdefault(bus, {}) 

51 names[pgn] = name if pgn not in names else None 

52 return buses

55def _migrate_j1939_frame_refs_to_names(documents: Documents) -> None: 

56 """ 

57 ``J1939FrameRef`` addresses its frame by ``frame_ref`` (name) or ``can_id`` instead of ``pgn``. 

58 

59 0.15 resolved a ``pgn`` on the ``bus_ref`` of the owning interface, where a PGN was unique, so the 

60 reference becomes the name of that frame. A ``pgn`` that names no frame, or several, did not load 

61 on 0.15 either; it is left in place so the load reports the reference. 

62 """ 

63 frames_by_bus = _j1939_frame_names_by_bus(documents) 

64 for uri, interface in _can_interfaces(documents): 

65 bus: Any = interface.get("bus_ref") 

66 names = frames_by_bus.get(bus, {}) if isinstance(bus, str) else {} 

67 for field in ("j1939_sender_frames", "j1939_receiver_frames"): 

68 for ref in _dicts(interface.get(field)): 

69 problem = _pgn_ref_to_name(ref, names) 

70 if problem is not None: 

71 logger.warning("%s: %s: %s on bus %r; the reference is left unchanged.", uri, field, problem, bus) 

72 

73 

74def _pgn_ref_to_name(ref: Node, names: dict[int, str | None]) -> str | None: 

75 """Replace the ``pgn`` of *ref* with the ``frame_ref`` it names in *names*; return why it cannot, if so.""" 

76 pgn = ref.get("pgn") 

77 if not _is_can_id(pgn): 

78 return None 

79 name = names.get(pgn) 

80 if name is None: 

81 return f"{'several J1939 frames share' if pgn in names else 'no J1939 frame has'} PGN {pgn:#x}" 

82 del ref["pgn"] 

83 ref.setdefault("frame_ref", name) 

84 return None
