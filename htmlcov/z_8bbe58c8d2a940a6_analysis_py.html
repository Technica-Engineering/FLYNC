1""" 

2Analyses over the variation points of a 150 % workspace: traceability and variant comparison. 

3 

4The functions here take the raw (safe-loaded) documents of a workspace, keyed by their workspace-relative ids, 

5and never touch the disk. 

6 

7- :func:`collect_variation_points` lists every ``variant_condition``: at the root of an ECU's ``ecu_metadata`` it 

8 decides which variants the ECU is part of, on an entry of the system topology which variants the entry is part of. 

9- :func:`trace_features` answers "which elements of the workspace does this feature decide?" - the link from a 

10 feature down to the realizing elements. 

11- :func:`compare_variants` answers "what is different between these two variants?" - in feature values and in 

12 the elements of the workspace. 

13""" 

14

23LABEL_KEYS = ("id", "name", "ecu_port") 

24"""Keys of a mapping whose value names the element in :meth:`VariationPoint.describe`, in order of preference."""

27@dataclass(frozen=True) 

28class VariationPoint: 

29 """ 

30 One variation point of the workspace: a ``variant_condition`` and the element it decides the presence of. 

31 

32 Attributes: 

33 document (str): Workspace-relative id of the document. 

34 location (YamlPath): Path to the element inside the document; empty for a condition at the document root, which 

35 decides the presence of the whole document (for an ECU's ``ecu_metadata``, of the whole ECU). 

36 condition (str | None): The condition, ``None`` when what is written is not a condition. 

37 label (str | None): The name of the element (its ``id``, ``name``, or ``ecu_port``), when it has one. 

38 """ 

39 

40 document: str 

41 location: YamlPath 

42 condition: str | None 

43 label: str | None = None 

44 

45 def describe(self) -> str: 

46 """A short human-readable reference, e.g. ``topology/ethernet_topology.flync.yaml: connections[1] (link_rear)``.""" 

47 

48 where = _format_location(self.location) 

49 text = f"{self.document}: {where}" if where else self.document 

50 return f"{text} ({self.label})" if self.label and where else text 

51 

52 def outcome(self, context: VariantContext) -> bool: 

53 """Whether the element is part of a variant.""" 

54 

55 return context.holds(self.condition)

58def _format_location(location: YamlPath) -> str: 

59 """Render a YAML path as ``a.b[0].c``.""" 

60 

61 text = "" 

62 for part in location: 

63 if isinstance(part, int): 

64 text += f"[{part}]" 

65 else: 

66 text += f".{part}" if text else str(part) 

67 return text

70def _condition_text(value: object) -> str | None: 

71 """The text of a condition, ``None`` when the value is not one (YAML booleans become ``true``/``false``).""" 

72 

73 if isinstance(value, bool): 

74 return str(value).lower() 

75 return value if isinstance(value, str) else None

78def _label(data: dict[Any, Any]) -> str | None: 

79 """The name of the element a mapping stands for, see :data:`LABEL_KEYS`.""" 

80 

81 for key in LABEL_KEYS: 

82 value = data.get(key) 

83 if isinstance(value, (str, int)) and not isinstance(value, bool): 

84 return str(value) 

85 return None

88def _walk(document: str, data: Any, path: YamlPath) -> Iterator[VariationPoint]: 

89 """Yield the variation points of a parsed YAML node: a mapping's own condition first, then those nested in it.""" 

90 

91 if isinstance(data, dict): 

92 if VARIANT_CONDITION_KEY in data: 

93 yield VariationPoint(document, path, _condition_text(data[VARIANT_CONDITION_KEY]), _label(data)) 

94 for key, value in data.items(): 

95 if key != VARIANT_CONDITION_KEY: 

96 yield from _walk(document, value, path + (key,)) 

97 elif isinstance(data, list): 

98 for index, item in enumerate(data): 

99 yield from _walk(document, item, path + (index,))

102def collect_variation_points(documents: Mapping[str, Any]) -> list[VariationPoint]: 

103 """ 

104 List every variation point of a workspace. 

105 

106 Args: 

107 documents (Mapping[str, Any]): The safe-loaded content of every document, by workspace-relative id. Documents in 

108 ``variability/`` are skipped. 

109 

110 Returns: 

111 list[VariationPoint]: The variation points in document order, a document's root condition first. 

112 """ 

113 

114 points: list[VariationPoint] = [] 

115 for uri in sorted(documents): 

116 if not is_in_variability(uri): 

117 points.extend(_walk(uri, documents[uri], ())) 

118 return points

121def _names(variability: Variability, condition: str | None) -> set[str]: 

122 """The features that ``condition`` refers to, qualifiers expanded; an unparsable condition refers to none.""" 

123 

124 if condition is None: 

125 return set() 

126 feature_model = variability.feature_model 

127 try: 

128 names = compile_condition(condition).names() 

129 except ConditionSyntaxError: 

130 return set() 

131 declared = {feature.name for feature in feature_model.features} 

132 return {name for name in feature_model.expand(names) if name in declared}

135def trace_features(variability: Variability, points: list[VariationPoint]) -> dict[str, list[VariationPoint]]: 

136 """ 

137 Find the variation points every feature decides, directly or through qualifiers. 

138 

139 Args: 

140 variability (Variability): The variability model. 

141 points (list[VariationPoint]): The variation points of the workspace, see :func:`collect_variation_points`. 

142 

143 Returns: 

144 dict[str, list[VariationPoint]]: For every feature (in declaration order), the variation points whose 

145 condition refers to it; features no variation point refers to map to an empty list. 

146 """ 

147 

148 traced: dict[str, list[VariationPoint]] = {feature.name: [] for feature in variability.feature_model.features} 

149 for point in points: 

150 for name in _names(variability, point.condition): 

151 traced[name].append(point) 

152 return traced

155@dataclass 

156class VariantComparison: 

157 """ 

158 The differences between two variants. 

159 

160 Attributes: 

161 left (str): Name of the first variant. 

162 right (str): Name of the second variant. 

163 features (list[tuple[str, object, object]]): ``(feature, left value, right value)`` for every feature that differs. 

164 elements (list[tuple[VariationPoint, bool, bool]]): ``(point, part of left, part of right)`` for every element that 

165 is part of one of the variants only. 

166 """ 

167 

168 left: str 

169 right: str 

170 features: list[tuple[str, object, object]] = field(default_factory=list) 

171 elements: list[tuple[VariationPoint, bool, bool]] = field(default_factory=list) 

172 

173 @property 

174 def identical(self) -> bool: 

175 """Whether the two variants resolve to the same workspace and feature values.""" 

176 

177 return not self.features and not self.elements

180def compare_variants(variability: Variability, points: list[VariationPoint], left: str, right: str) -> VariantComparison: 

181 """ 

182 Compare two variants: their feature values and the elements of the workspace that differ. 

183 

184 Args: 

185 variability (Variability): The variability model. 

186 points (list[VariationPoint]): The variation points of the workspace, see :func:`collect_variation_points`. 

187 left (str): Name of the first variant. 

188 right (str): Name of the second variant. 

189 

190 Returns: 

191 VariantComparison: The differences, in feature declaration order and document order. 

192 

193 Raises: 

194 KeyError: If one of the variants is not declared. 

195 """ 

196 

197 left_context, right_context = variability.context(left), variability.context(right) 

198 comparison = VariantComparison(left, right) 

199 for name in (feature.name for feature in variability.feature_model.features): 

200 left_value, right_value = left_context.assignment.get(name), right_context.assignment.get(name) 

201 if left_value != right_value: 

202 comparison.features.append((name, left_value, right_value)) 

203 for point in points: 

204 left_outcome, right_outcome = point.outcome(left_context), point.outcome(right_context) 

205 if left_outcome != right_outcome: 

206 comparison.elements.append((point, left_outcome, right_outcome)) 

207 return comparison
