1""" 

2``flync migrate`` — bring a FLYNC workspace up to the installed FLYNC schema. 

3 

4Detects the version gap between the workspace's recorded version and the installed FLYNC release and 

5applies the registered schema migrations in order, recording them in ``.flync/migrations/history.yaml``. 

6""" 

7 

8from __future__ import annotations 

9 

10from pathlib import Path 

11 

12import typer 

13from rich.console import Console 

14from rich.table import Table 

15from typing_extensions import Annotated

17from flync.core.version_migrators.engine import Migrator 

18from flync.core.version_migrators.engine import WorkspaceMigrationState as State 

19 

20console = Console(force_terminal=True, legacy_windows=False) 

21app = typer.Typer(help="Migrate a FLYNC workspace to the installed FLYNC schema.")

24def _print_plan(migrator: Migrator, path: Path) -> State | None: 

25 """Print the migration status/plan for ``path`` and return its state.""" 

26 try: 

27 status = migrator.plan(path) 

28 except Exception as exc: # pragma: no cover - defensive 

29 console.print(f"[bold red]Could not read migration status: {exc}[/bold red]") 

30 return None 

31 

32 # NO_VERSION is the only state without a recorded stamp; handle it first so the 

33 # narrowing assert below never runs on a ``None`` recorded version. 

34 if status.state is State.NO_VERSION: 

35 console.print( 

36 "[bold yellow]Workspace has no recorded FLYNC version[/bold yellow]\n" 

37 f"`.flync/config.yaml` at {path} is missing or has no `version` entry, so FLYNC treats the workspace as " 

38 "already in the installed schema.\n" 

39 f"Run `flync migrate --init {path}` to record the current version (no data is changed)." 

40 ) 

41 return status.state 

42 

43 assert status.recorded is not None 

44 if status.state is State.NEWER: 

45 console.print( 

46 f"[bold yellow]Workspace was created by FLYNC {status.recorded.version}, newer than the installed " 

47 f"{status.current.version}.[/bold yellow]\nRefusing to migrate a workspace from a newer FLYNC." 

48 ) 

49 elif status.state is State.UNSUPPORTED_OLD: 

50 console.print( 

51 f"[bold red]Workspace records FLYNC {status.recorded.version}, older than the lowest supported " 

52 f"{status.lowest_supported}.[/bold red]\nThere is no migration path; re-create or re-import the workspace." 

53 ) 

54 elif status.state is State.PENDING_MIGRATION: 

55 table = Table(title=f"Migrations to apply: {status.recorded.version} → {status.current.version}", show_lines=True) 

56 table.add_column("Migration", style="cyan") 

57 table.add_column("Target", style="green") 

58 table.add_column("Description", overflow="fold") 

59 for migration in status.pending: 

60 table.add_row(migration.unique_id, migration.target_version, migration.description) 

61 console.print(table) 

62 console.print(f"\nRun `flync migrate {path}` to apply {len(status.pending)} migration(s).") 

63 elif status.requires_stamp: 

64 console.print( 

65 f"[green]Workspace is up to date, but its version stamp ({status.recorded.version}) lags the installed " 

66 f"{status.current.version}.[/green]\nRun `flync migrate {path}` to advance the stamp." 

67 ) 

68 else: 

69 console.print(f"[green]Workspace is up to date (FLYNC {status.current.version}).[/green]") 

70 return status.state

73@app.command(help="Migrate a FLYNC workspace to the installed FLYNC schema.") 

74def migrate( 

75 path: Annotated[str, typer.Argument(help="Path to the FLYNC workspace directory.")], 

76 dry_run: Annotated[ 

77 bool, 

78 typer.Option("--dry-run", "--plan", help="Show what would run without changing anything."), 

79 ] = False, 

80 init: Annotated[ 

81 bool, 

82 typer.Option("--init", help="Only stamp the workspace with the current FLYNC version (no transforms)."), 

83 ] = False, 

84): 

85 """ 

86 Apply pending migrations to the workspace at ``path``. 

87 

88 Without flags, applies pending migrations in order and records them in 

89 ``.flync/migrations/history.yaml``. ``--dry-run`` only prints the plan; ``--init`` only stamps a 

90 baseline version without touching the workspace content. 

91 

92 Exit codes: 0 on success or plan-only, 1 when an unsupported/error state prevents migration. 

93 """ 

94 resolved = Path(path).resolve() 

95 if not resolved.exists(): 

96 console.print(f"[bold red]Path does not exist: {path}[/bold red]") 

97 raise typer.Exit(code=1) 

98 

99 migrator = Migrator() 

100 

101 if init: 

102 try: 

103 migrator.stamp(resolved) 

104 except Exception as exc: # pragma: no cover - defensive 

105 console.print(f"[bold red]Could not stamp the workspace: {exc}[/bold red]") 

106 raise typer.Exit(code=1) 

107 console.print(f"[green]Stamped {resolved} with the current FLYNC version.[/green]") 

108 raise typer.Exit(code=0) 

109 

110 state = _print_plan(migrator, resolved) 

111 

112 if dry_run: 

113 raise typer.Exit(code=0 if state is not State.UNSUPPORTED_OLD else 1) 

114 if state in (State.NO_VERSION, State.NEWER, State.UNSUPPORTED_OLD): 

115 raise typer.Exit(code=0 if state is State.NEWER else 1) 

116 

117 try: 

118 result = migrator.run(resolved) 

119 except RuntimeError as exc: 

120 console.print(f"[bold red]{exc}[/bold red]") 

121 raise typer.Exit(code=1) 

122 

123 console.print(f"[bold green]{result.message}[/bold green]") 

124 for uri in result.changed_uris: 

125 console.print(f" [cyan]updated[/cyan] {uri}") 

126 if result.changed_uris: 

127 console.print("Recorded in .flync/migrations/history.yaml.") 

128 raise typer.Exit(code=0) 

129 

130 

131if __name__ == "__main__": 

132 app()
