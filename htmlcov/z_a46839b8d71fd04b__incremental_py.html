28 def update_document(self, uri: PathType) -> list[str]: 

29 """ 

30 Incrementally reconcile the workspace with a single document that changed on disk. 

31 

32 Three cases are handled in place, all re-validating the affected node's ancestor spine up to the 

33 root while reusing the already-loaded sibling models (so cross-document references, resolved by 

34 ancestor validators, stay correct) without a full :meth:`load_workspace`: 

35 

36 - **changed** (known document, still on disk): reload just that node and its subtree. 

37 - **added** (unknown document, now on disk): reload the nearest existing ancestor, whose 

38 directory scan picks up the new file. 

39 - **removed** (known document, gone from disk): reload the parent, whose scan drops the file. 

40 

41 New or removed items directly under the root, and any change that breaks a root-resolved 

42 reference, fall back to :meth:`_revalidate_all` (a full re-validation from the cached ASTs, no 

43 disk re-read). An unexpected failure falls back to a from-disk :meth:`_reset_and_reload`. 

44 

45 Args: 

46 uri (PathType): Path of the affected document, absolute or workspace-relative. 

47 

48 Returns: 

49 list[str]: The ids of every document whose model or diagnostics were recomputed. 

50 """ 

51 

52 uri = Document.normalize_uri(uri, self.workspace_root) 

53 target = self.workspace_root / uri if self.workspace_root else Path(uri) 

54 node = self._doc_index.get(uri) 

55 exists = target.exists() 

56 try: 

57 if node is not None and exists: 

58 result = self._reload_document(node, uri) 

59 elif node is None and exists: 

60 result = self._add_document(uri) 

61 elif node is not None: 

62 result = self._remove_document(node) 

63 else: 

64 result = self._reset_and_reload() 

65 return result 

66 except Exception: 

67 logger.exception("update_document: partial update of %s failed, reloading workspace", uri) 

68 return self._reset_and_reload()

70 def _add_document(self, uri: str) -> list[str]: 

71 """ 

72 Bring a newly created document into the workspace. 

73 

74 The nearest existing ancestor load-node is reloaded; its directory scan picks up the new file and 

75 the spine above it is re-validated. A new item directly under the root (e.g. a whole new ECU) 

76 has no partial ancestor, so it falls back to a full reload. 

77 

78 Args: 

79 uri (str): Workspace-relative id of the new document. 

80 

81 Returns: 

82 list[str]: Ids of every document that was recomputed. 

83 """ 

84 

85 ancestor = self._nearest_ancestor_node(uri) 

86 if ancestor is None or ancestor.link is None: 

87 return self._revalidate_all() 

88 return self._reload_from_node(ancestor)

90 def _remove_document(self, node: LoadNode) -> list[str]: 

91 """ 

92 Drop a document that no longer exists on disk. 

93 

94 The parent load-node is reloaded; its directory scan no longer sees the file, so the item drops 

95 out of the model, and the spine above is re-validated. Removing the root falls back to a full 

96 reload. 

97 

98 Args: 

99 node (LoadNode): The indexed node for the removed document. 

100 

101 Returns: 

102 list[str]: Ids of every document that was recomputed. 

103 """ 

104 

105 if node.link is None: 

106 return self._revalidate_all() 

107 parent = self._doc_index.get(self.document_id_from_path(str(node.link.parent_path))) 

108 if parent is None: 

109 return self._revalidate_all() 

110 return self._reload_from_node(parent)

112 def _nearest_ancestor_node(self, uri: str) -> Optional[LoadNode]: 

113 """ 

114 Return the closest indexed load-node that contains ``uri`` in its directory tree. 

115 

116 Walks up the path parents until an indexed node is found (the root always matches once loaded). 

117 

118 Args: 

119 uri (str): Workspace-relative id to locate. 

120 

121 Returns: 

122 LoadNode | None: The nearest ancestor node, or ``None`` if none is indexed. 

123 """ 

124 

125 current = Path(uri).parent 

126 while True: 

127 doc_id = current.as_posix() 

128 ancestor = self._doc_index.get(doc_id) 

129 if ancestor is not None: 

130 return ancestor 

131 if doc_id in (".", ""): 

132 return None 

133 current = current.parent

135 def _reload_document(self, node: LoadNode, uri: str) -> list[str]: 

136 """ 

137 Perform the in-place reload of ``node`` and re-validate its ancestor spine. 

138 

139 Args: 

140 node (LoadNode): The indexed node for the changed document. 

141 uri (str): Workspace-relative id of the changed document. 

142 

143 Returns: 

144 list[str]: Ids of all documents that were recomputed (leaf subtree + ancestors). 

145 """ 

146 

147 text = read_file(self.workspace_root / uri) # type: ignore[operator] 

148 if uri in self.documents: 

149 self.documents[uri].update_text(text) 

150 else: 

151 doc = Document(uri, text, self.configuration.map_objects) 

152 doc.parse() 

153 self.documents[uri] = doc 

154 

155 return self._reload_from_node(node)

157 def _reload_from_node(self, node: LoadNode) -> list[str]: 

158 """ 

159 Reload ``node``'s subtree and re-validate every ancestor up to the root. 

160 

161 Shared by every partial-update flavour: a changed document reloads its own node, while a new or 

162 removed document reloads the nearest surviving ancestor (whose directory scan then picks up or 

163 drops the affected file). 

164 

165 Args: 

166 node (LoadNode): The node whose subtree is reloaded first. 

167 

168 Returns: 

169 list[str]: Ids of all documents that were recomputed. 

170 """ 

171 

172 affected: list[str] = [] 

173 cur_child, ids = self._reload_subtree(node) 

174 affected += ids 

175 

176 cur_link = self._doc_index[node.doc_id].link 

177 while cur_link is not None: 

178 parent_id = self.document_id_from_path(str(cur_link.parent_path)) 

179 parent_node = self._doc_index.get(parent_id) 

180 if parent_node is None: 

181 raise ValueError(f"missing parent node for {cur_link.parent_path}") 

182 rebuilt = self._rebuild_ancestor(parent_node, cur_link, cur_child) 

183 if rebuilt is None: 

184 # Reusing sibling instances is cheap but leaves error-recovery unable to prune a bad 

185 # value out of a reused child. When that happens, re-validate this ancestor's whole 

186 # subtree (from the cached ASTs, no disk read) so the policy pruning behaves as it does 

187 # on a full load. The root's subtree is the whole workspace, so revalidate everything. 

188 if parent_node.link is None: 

189 return self._revalidate_all() 

190 rebuilt, ids = self._reload_subtree(parent_node) 

191 affected += ids 

192 affected.append(parent_id) 

193 cur_child = rebuilt 

194 cur_link = self._doc_index[parent_id].link 

195 self.flync_model = cur_child # type: ignore[assignment] 

196 

197 return list(dict.fromkeys(affected))

199 def _reload_subtree(self, node: LoadNode) -> tuple[object, list[str]]: 

200 """ 

201 Drop and re-load ``node`` together with everything indexed underneath it. 

202 

203 Args: 

204 node (LoadNode): Root of the subtree to reload. 

205 

206 Returns: 

207 tuple[object, list[str]]: The reloaded model value and the ids of the documents touched. 

208 """ 

209 

210 before = [sub.doc_id for sub in self._subtree_nodes(node)] 

211 for doc_id in before: 

212 self.documents_diags.pop(doc_id, None) 

213 if doc_id != node.doc_id: 

214 self._doc_index.pop(doc_id, None) 

215 # drop a cached document whose file has been deleted so the rescan does not resurrect it 

216 if doc_id in self.documents and not (self.workspace_root / doc_id).exists(): # type: ignore[operator] 

217 self.documents.pop(doc_id, None) 

218 if self.configuration.map_objects: 

219 self._purge_object_subtree(node.object_paths) 

220 model = self._load_from_path( 

221 node.path, 

222 node.current_type, 

223 node.current_type_name, 

224 list(node.object_paths), 

225 node.link, 

226 ) 

227 # union with the post-reload subtree so newly added documents are reported too 

228 after = [sub.doc_id for sub in self._subtree_nodes(node)] 

229 return model, list(dict.fromkeys(before + after))

231 def _rebuild_ancestor(self, parent_node: LoadNode, link: ParentLink, new_child): 

232 """ 

233 Re-validate ``parent_node`` with its changed child replaced, reusing every other child instance. 

234 

235 The parent's own inline file is re-read but the unchanged external children are taken straight 

236 from the previously loaded parent instance, so validation only re-runs the parent's own 

237 validators (including reference binding) instead of re-parsing sibling documents. 

238 

239 Args: 

240 parent_node (LoadNode): The ancestor being rebuilt. 

241 link (ParentLink): How ``new_child`` attaches to this ancestor. 

242 new_child: The freshly loaded value for ``link.field_name``. 

243 

244 Returns: 

245 The rebuilt (and parent-normalized) ancestor model. 

246 """ 

247 

248 old_parent = parent_node.model 

249 parent_type = parent_node.current_type 

250 new_branch = self._splice_branch(old_parent, link, new_child) 

251 module_load_info = self._ancestor_load_info(parent_node, old_parent, link, new_branch) 

252 

253 new_parent = self._validate_node(parent_node, module_load_info) 

254 if self.configuration.map_objects and old_parent is not None: 

255 if new_parent is not None: 

256 self._remap_ancestor_objects(parent_type, old_parent, new_parent) 

257 else: 

258 self._detach_object(parent_node) 

259 return new_parent

261 def _ancestor_load_info(self, parent_node: LoadNode, old_parent, link: ParentLink, new_branch) -> dict: 

262 """ 

263 Gather the field data to re-validate ``parent_node`` with ``link.field_name`` replaced. 

264 

265 Unchanged external children are reused straight from ``old_parent`` (so they are not re-read or 

266 re-validated), implied fields are recomputed from the path, and the parent's own inline file is 

267 merged back in. 

268 """ 

269 

270 module_load_info: dict = {} 

271 for field_name, field_info in parent_node.current_type.model_fields.items(): 

272 if field_name == link.field_name: 

273 module_load_info[field_name] = new_branch 

274 continue 

275 if get_metadata(field_info.metadata, External) is not None: 

276 value = getattr(old_parent, field_name, None) 

277 if value is not None: 

278 module_load_info[field_name] = value 

279 continue 

280 implied = get_metadata(field_info.metadata, Implied) 

281 if implied is not None: 

282 self._handle_implied_field_load(parent_node.path, module_load_info, field_name, implied) 

283 self._append_to_info_dict(parent_node.path, module_load_info) 

284 return module_load_info

286 def _detach_object(self, node: LoadNode) -> None: 

287 """Clear the object-map entry of a node whose model failed to rebuild, so it holds no stale model.""" 

288 if not node.object_paths: 

289 return 

290 oid = ObjectId(node.object_paths[0].strip(".")) 

291 semantic = self.objects.get(oid) 

292 if semantic is None or semantic.model is None: 

293 return 

294 self._unmap_object_id(semantic.model, oid) 

295 cast(SemanticObject, semantic).model = None # type: ignore[assignment]

298 def _splice_branch(old_parent, link: ParentLink, new_child): 

299 """Build the new value for ``link.field_name`` with ``new_child`` put in the right slot.""" 

300 if link.container == "list": 

301 new_list = list(getattr(old_parent, link.field_name, None) or []) 

302 if isinstance(link.key, int) and 0 <= link.key < len(new_list): 

303 new_list[link.key] = new_child 

304 else: 

305 new_list.append(new_child) 

306 return new_list 

307 if link.container == "dict": 

308 new_dict = dict(getattr(old_parent, link.field_name, None) or {}) 

309 new_dict[link.key] = new_child 

310 return new_dict 

311 return new_child

313 def _subtree_nodes(self, node: LoadNode) -> list[LoadNode]: 

314 """Return ``node`` and every indexed node whose object path sits underneath it.""" 

315 own = {p.strip(".") for p in node.object_paths} 

316 own.discard("") 

317 prefixes = tuple(f"{p}." for p in own) 

318 result = [] 

319 for candidate in self._doc_index.values(): 

320 ids = {p.strip(".") for p in candidate.object_paths} 

321 if ids & own or (prefixes and any(cid.startswith(prefixes) for cid in ids)): 

322 result.append(candidate) 

323 return result

325 def _purge_object_subtree(self, object_paths: list[str]) -> None: 

326 """ 

327 Remove all object-map entries under ``object_paths`` so a reload can register them fresh. 

328 

329 The node's own ids are dropped from :attr:`objects`/:attr:`sources` (the reload re-creates them 

330 with the new model) but kept in the parent's child index, since the parent is not reloaded and 

331 would otherwise lose the edge to this node. 

332 """ 

333 

334 own = {p.strip(".") for p in object_paths} 

335 own.discard("") 

336 prefixes = tuple(f"{p}." for p in own) 

337 

338 def in_subtree(oid: str) -> bool: 

339 """Utility function to check if object id is in the subtree.""" 

340 return oid in own or (bool(prefixes) and oid.startswith(prefixes)) 

341 

342 def is_descendant(oid: str) -> bool: 

343 """Utility function to check if object id is descendant in the path.""" 

344 return bool(prefixes) and oid.startswith(prefixes) 

345 

346 # Use self.objects keys directly — list_objects() includes virtual 

347 # aliases from _duplicated_objects_ids.values() that are not actual keys 

348 # in self.objects and would cause a KeyError on pop. 

349 real_objs = [o for o in self.objects if in_subtree(str(o))] 

350 for oid in real_objs: 

351 semantic = self.objects.pop(oid) 

352 self.sources.pop(oid, None) 

353 self._unmap_object_id(semantic.model, oid) 

354 if self._duplicated_objects_ids: 

355 for oid in [o for o in self.list_objects() if in_subtree(str(o))]: 

356 self._duplicated_objects_ids.pop(oid, None) 

357 

358 for pid in [p for p in self._children_by_parent if in_subtree(p)]: 

359 self._children_by_parent.pop(pid, None) 

360 for parent_id, children in self._children_by_parent.items(): 

361 self._children_by_parent[parent_id] = [c for c in children if not is_descendant(c)] 

362 self._linked_child_ids = {c for c in self._linked_child_ids if not is_descendant(c)}

364 def _remap_ancestor_objects(self, parent_type, old_parent, new_parent) -> None: 

365 """Point the ancestor's (and its external container fields') object entries at the rebuilt instances.""" 

366 self._remap_model_ids(old_parent, new_parent) 

367 for field_name, field_info in parent_type.model_fields.items(): 

368 if get_metadata(field_info.metadata, External) is None: 

369 continue 

370 old_value = getattr(old_parent, field_name, None) 

371 new_value = getattr(new_parent, field_name, None) 

372 if isinstance(old_value, (list, dict)) and old_value is not new_value: 

373 self._remap_model_ids(old_value, new_value)

375 def _unmap_object_id(self, model, oid: ObjectId) -> None: 

376 """Remove a single object id from the ``model -> ids`` reverse index.""" 

377 if model is None: 

378 return 

379 ids = self._model_to_object_ids.get(id(model)) 

380 if ids is None: 

381 return 

382 remaining = [i for i in ids if i != oid] 

383 if remaining: 

384 self._model_to_object_ids[id(model)] = remaining 

385 else: 

386 self._model_to_object_ids.pop(id(model), None) 

387 

388 def _remap_model_ids(self, old_model, new_model) -> None: 

389 """Move the object ids registered for ``old_model`` onto ``new_model``.""" 

390 ids = self._model_to_object_ids.pop(id(old_model), None) 

391 if not ids: 

392 return 

393 for oid in ids: 

394 semantic = self.objects.get(oid) 

395 if semantic is not None: 

396 semantic.model = new_model 

397 self._model_to_object_ids[id(new_model)] = ids

410 def _revalidate_all(self) -> list[str]: 

411 """ 

412 Rebuild the whole model and object graph from the already-parsed documents. 

413 

414 Used when a partial update cannot be completed in place (e.g. a change breaks a root-resolved 

415 reference, or a new item lands directly under the root). Unlike :meth:`_reset_and_reload` it does 

416 not re-read or re-parse files through the process pool - it reuses the in-memory document ASTs 

417 that ``update_document`` keeps in sync, reading from disk only files not seen before (a newly 

418 created document). This makes the fallback markedly cheaper than a cold load. 

419 """ 

420 

421 self._clear_derived_state() 

422 self.flync_model = self._load_from_path(self.workspace_root) # type: ignore[arg-type] 

423 return list(self.documents_diags.keys())

425 def _reset_and_reload(self) -> list[str]: 

426 """Clear all derived state and reload the workspace from disk (the last-resort safety path).""" 

427 self.documents.clear() 

428 self._clear_derived_state() 

429 self._open_documents() 

430 self.flync_model = self._load_from_path(self.workspace_root) # type: ignore[arg-type] 

431 return list(self.documents_diags.keys())
