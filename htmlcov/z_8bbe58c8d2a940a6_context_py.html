1""" 

2The configuration of one product variant and its evaluation context. 

3 

4A variant states only its own decisions. :func:`configure` completes them into a full configuration: 

5 

61. The decisions written for the variant itself. 

72. The decisions it inherits through ``extends``, nearer variants overriding farther ones. 

83. The default of every feature that is still undecided (``false`` for a boolean feature without a default). 

9 

10An enumerated feature without a value and without a default is recorded as an open decision; 

11:class:`~flync.model.flync_4_variability.variant.Variability` reports it. 

12 

13A :class:`VariantContext` holds the completed configuration of a variant and evaluates variant conditions against 

14it. The workspace loader uses it to resolve the variation points of the 150 % workspace, and a loaded 

15:class:`~flync.model.flync_model.FLYNCModel` exposes it as ``active_variant`` so generators can tell which variant 

16they hold, where each value comes from, and read the attributes of its features. 

17 

18The functions here only read the variability model, so they work on a partially valid model too. 

19"""

35class ValueSource(StrEnum): 

36 """ 

37 Where the value of a feature in a variant comes from. 

38 

39 - ``assigned``: written in the variant itself. 

40 - ``inherited``: written in a variant it ``extends``. 

41 - ``default``: the feature's default, or ``false`` for an undecided boolean feature. 

42 """ 

43 

44 ASSIGNED = "assigned" 

45 INHERITED = "inherited" 

46 DEFAULT = "default"

65def normalize_value(value: object, is_boolean: bool) -> ConditionValue | None: 

66 """Read an assigned value the way conditions compare it, or return ``None`` when it does not fit the feature.""" 

67 

68 if is_boolean: 

69 return value if isinstance(value, bool) else None 

70 if isinstance(value, (str, int, float)) and not isinstance(value, bool): 

71 return str(value) 

72 return None

75def configure( 

76 feature_model: FeatureModel, 

77 assigned: Mapping[str, object], 

78 inherited: Mapping[str, object] | None = None, 

79) -> Configuration: 

80 """ 

81 Complete the decisions of a variant into a full configuration. 

82 

83 Decisions that name an undeclared feature or do not fit its type are skipped here; the variability model reports them. 

84 

85 Args: 

86 feature_model (FeatureModel): The features of the product line. 

87 assigned (Mapping[str, object]): The values written in the variant itself. 

88 inherited (Mapping[str, object] | None): The values it inherits through ``extends`` (nearer variants already merged). 

89 

90 Returns: 

91 Configuration: The values with their sources, and the enumerated features left undecided. 

92 """ 

93 

94 config = Configuration() 

95 layers = ((assigned, ValueSource.ASSIGNED), (inherited or {}, ValueSource.INHERITED)) 

96 for feature in feature_model.features: 

97 decided = _decided_value(feature, layers) 

98 if decided is None: 

99 decided = _default_value(feature) 

100 if decided is None: 

101 config.open_decisions.append(feature.name) 

102 else: 

103 config.values[feature.name], config.sources[feature.name] = decided 

104 return config

107def _decided_value(feature: Feature, layers: tuple[tuple[Mapping[str, object], ValueSource], ...]) -> tuple[ConditionValue, ValueSource] | None: 

108 """The first decision for ``feature`` that fits it, searching the variant's own decisions before inherited ones.""" 

109 

110 for decisions, source in layers: 

111 if feature.name in decisions: 

112 value = normalize_value(decisions[feature.name], feature.is_boolean) 

113 if value is not None: 

114 return value, source 

115 return None 

116 

117 

118def _default_value(feature: Feature) -> tuple[ConditionValue, ValueSource] | None: 

119 """The default of ``feature`` (``false`` for a boolean feature without one), or ``None`` when it has none.""" 

120 

121 if feature.default is not None: 

122 value = normalize_value(feature.default, feature.is_boolean) 

123 if value is not None: 

124 return value, ValueSource.DEFAULT 

125 return (False, ValueSource.DEFAULT) if feature.is_boolean else None

128@dataclass(frozen=True) 

129class VariantContext: 

130 """ 

131 The complete configuration of one product variant. 

132 

133 Attributes: 

134 name (str): Name of the variant. 

135 assignment (Mapping[str, ConditionValue]): Value of every feature that has one in this variant: ``True``/``False`` for a 

136 boolean feature, the selected value (text) for an enumerated feature. 

137 symbols (SymbolTable): Features and qualifiers of the feature model. 

138 description (str | None): Description of the variant. 

139 feature_attributes (Mapping[str, Mapping[str, AttributeValue]]): Attributes of every feature that declares some. 

140 sources (Mapping[str, ValueSource]): Where the value of each feature comes from. 

141 """ 

142 

143 name: str 

144 assignment: Mapping[str, ConditionValue] 

145 symbols: SymbolTable = field(repr=False) 

146 description: str | None = None 

147 feature_attributes: Mapping[str, Mapping[str, AttributeValue]] = field(default_factory=dict, repr=False) 

148 sources: Mapping[str, ValueSource] = field(default_factory=dict, repr=False)

152 def truth(self, name: str) -> bool: 

153 """Truth value of a boolean feature or a qualifier.""" 

154 

155 qualifier = self.symbols.qualifiers.get(name) 

156 if qualifier is not None: 

157 return qualifier.evaluate(self) 

158 return self.assignment[name] is True 

159 

160 def value_of(self, name: str) -> ConditionValue: 

161 """Value of a feature, or the truth value of a qualifier.""" 

162 

163 if name in self.symbols.qualifiers: 

164 return self.truth(name) 

165 return self.assignment[name] 

166 

167 def rank(self, name: str, value: ConditionValue) -> int: 

168 """Position of ``value`` among the declared values of the ordered feature ``name``.""" 

169 

170 values = self.symbols.features[name].values or () 

171 return values.index(str(value))

195 def is_enabled(self, name: str) -> bool: 

196 """ 

197 Return whether a boolean feature or a qualifier is true in this variant. 

198 

199 Raises: 

200 KeyError: If ``name`` is neither a boolean feature nor a qualifier.

202

207 def evaluate(self, condition: str | bool | Condition) -> bool: 

208 """ 

209 Evaluate a variant condition in this variant. 

210 

211 Raises: 

212 ConditionSyntaxError: If the condition does not parse. 

213 KeyError: If the condition uses a name that is not declared. 

214 ValueError: If the condition compares an ordered feature with a value it does not declare. 

215 """ 

216 

217 compiled = condition if isinstance(condition, Condition) else compile_condition(condition) 

218 return compiled.evaluate(self) 

219 

220 def holds(self, condition: object) -> bool: 

221 """ 

222 Evaluate a variant condition, treating an invalid condition (or a value that is not a condition at all) as false. 

223 

224 Used where a broken condition is reported separately, so the element it guards is simply not part of the 

225 variant instead of failing the whole load. 

226 """ 

227 

228 try: 

229 compiled = condition if isinstance(condition, Condition) else compile_condition(condition) 

230 return compiled.evaluate(self) 

231 except (ValueError, KeyError, TypeError, IndexError, RecursionError): 

232 return False

241def extends_chain(variability: Variability, name: str) -> list[str]: 

242 """ 

243 Return ``name`` followed by the variants it inherits from, nearest first. 

244 

245 The chain stops at a variant that does not exist or that was already visited, so it is safe on an invalid model. 

246 """ 

247 

248 variants = _variant_by_name(variability) 

249 chain: list[str] = [] 

250 current: str | None = name 

251 while current is not None and current in variants and current not in chain: 

252 chain.append(current) 

253 current = variants[current].extends 

254 return chain

257def raw_assignment(variability: Variability, name: str) -> dict[str, object]: 

258 """Return the assignments written for variant ``name`` and its ``extends`` chain, nearer variants overriding.""" 

259 

260 variants = _variant_by_name(variability) 

261 merged: dict[str, object] = {} 

262 for current in reversed(extends_chain(variability, name)): 

263 merged.update(variants[current].features) 

264 return merged

267def inherited_assignment(variability: Variability, name: str) -> dict[str, object]: 

268 """Return the assignments variant ``name`` inherits through ``extends`` (without its own), nearer variants overriding.""" 

269 

270 variants = _variant_by_name(variability) 

271 merged: dict[str, object] = {} 

272 for current in reversed(extends_chain(variability, name)[1:]): 

273 merged.update(variants[current].features) 

274 return merged

278 """

291def build_context(variability: Variability, name: str) -> VariantContext: 

292 """ 

293 Build the evaluation context of variant ``name``. 

294 

295 Args: 

296 variability (Variability): The variability model of the workspace. 

297 name (str): Name of the variant. 

298 

299 Returns: 

300 VariantContext: The variant with its complete configuration. 

301 

302 Raises: 

303 KeyError: If the variant is not declared. 

304 """ 

305 

306 variants = _variant_by_name(variability) 

307 if name not in variants: 

308 raise KeyError(name) 

309 feature_model = variability.feature_model 

310 configuration = variant_configuration(variability, name) 

311 return VariantContext( 

312 name=name, 

313 assignment=MappingProxyType(dict(configuration.values)), 

314 symbols=feature_model.symbol_table(), 

315 description=variants[name].description, 

316 feature_attributes=MappingProxyType( 

317 {feature.name: MappingProxyType(dict(feature.attributes)) for feature in feature_model.features if feature.attributes} 

318 ), 

319 sources=MappingProxyType(dict(configuration.sources)), 

320 )
