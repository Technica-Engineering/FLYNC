154def convert(ctx, **kwargs): 

155 """Convert one or many sources to one or many destinations. 

156 

157 \b 

158 flync-converter convert \\ 

159 source -p first/ -f flync \\ 

160 source -p buses/ -f dbc --baud-rate-default 500000 \\ 

161 destination -p merged/ -f json 

162 

163 Every source is decoded to a FLYNC model, the models are merged in the 

164 order given, and the merged model is encoded to every destination. Run 

165 ``convert source -f <format> --help`` to list that format's own options. 

166 

167 The flat ``-s/-o/-sf/-of`` form with ``--src-<field>`` / ``--dst-<field>`` 

168 options is deprecated and runs a one-to-one conversion. 

169 

170 \f 

171 The group itself does nothing: it also runs ahead of a subcommand's 

172 ``--help``. The conversion runs in :func:`_run_convert`. 

173 """

241@click.pass_context 

242def _run_convert(ctx, parsed: list[_ParsedEndpoint], on_conflict, fail_fast, dry_run, no_preview, **kwargs): 

243 """Run the conversion described by the ``source`` / ``destination`` subcommands or by the flat form.""" 

244 flat = _flat_options_given(ctx) 

245 if flat and parsed: 

246 raise click.UsageError(f"{', '.join(flat)} cannot be combined with source/destination subcommands.", ctx) 

247 if flat and dry_run: 

248 raise click.UsageError("--dry-run needs source/destination subcommands; the flat -s/-o form does not support it.", ctx) 

249 if flat: 

250 click.echo( 

251 "Warning: the flat -s/-o form of convert is deprecated. Use: convert source -p <path> -f <format> destination -p <path> -f <format>", 

252 err=True, 

253 ) 

254 _convert_flat(ctx, no_preview, **kwargs) 

255 return 

256 if not parsed: 

257 click.echo(ctx.get_help()) 

258 ctx.exit(2) 

259 

260 roles = [item.role for item in parsed] 

261 if "destination" in roles and "source" in roles[roles.index("destination") :]: 

262 raise click.UsageError("Sources must come before destinations.", ctx) 

263 if "source" not in roles: 

264 raise click.UsageError("At least one source is required.", ctx) 

265 if "destination" not in roles: 

266 raise click.UsageError("At least one destination is required.", ctx) 

267 

268 sources = [item for item in parsed if item.role == "source"] 

269 destinations = [item for item in parsed if item.role == "destination"] 

270 _convert_chained(ctx, sources, destinations, OnConflict(on_conflict), fail_fast, dry_run, no_preview)

273def _flat_options_given(ctx: click.Context) -> list[str]: 

274 """Return the flags of the deprecated flat form given on the command line, in declaration order.""" 

275 command = cast(ConvertGroup, ctx.command) 

276 flat_names = {"source", "output", "source_format", "output_format", "source_config_file", "destination_config_file"} 

277 return [ 

278 param.opts[0] 

279 for param in command.params 

280 if param.name is not None 

281 and (param.name in flat_names or param.name.startswith((SRC_PREFIX, DST_PREFIX))) 

282 and ctx.get_parameter_source(param.name) not in (None, ParameterSource.DEFAULT) 

283 ]

286def _convert_flat( 

287 ctx: click.Context, 

288 no_preview: bool, 

289 source=None, 

290 output=None, 

291 source_format="flync", 

292 output_format="flync", 

293 source_config_file=None, 

294 destination_config_file=None, 

295 **kwargs, 

296) -> None: 

297 """Run the deprecated flat one-to-one form. 

298 

299 Args: 

300 ctx: Parsed Click context of ``convert``. 

301 no_preview: Hide the conversion preview (``--no-preview``). 

302 source: Source folder path, prompted when omitted. 

303 output: Output folder path, prompted when omitted. 

304 source_format: Source format (default: flync). 

305 output_format: Destination format (default: flync). 

306 source_config_file: Optional source configuration YAML (``--src-config``). 

307 ``--src-<field>`` options override the values in it. 

308 destination_config_file: Optional destination configuration YAML (``--dst-config``). 

309 ``--dst-<field>`` options override the values in it. 

310 **kwargs: The ``--src-<field>`` / ``--dst-<field>`` options. 

311 """ 

312 if source is None: 

313 source = click.prompt("source location") 

314 if output is None: 

315 output = click.prompt("output location") 

316 if source_format == output_format: 

317 click.echo("Source and output formats are the same. No conversion needed.") 

318 return 

319 

320 src_fields = {k[len(SRC_PREFIX) :]: v for k, v in kwargs.items() if k.startswith(SRC_PREFIX) and v is not None} 

321 dst_fields = {k[len(DST_PREFIX) :]: v for k, v in kwargs.items() if k.startswith(DST_PREFIX) and v is not None} 

322 

323 command = cast(ConvertGroup, ctx.command) 

324 sources = [_cli_endpoint(ctx, command, SRC_PREFIX, source_format, source, source_config_file, src_fields)] 

325 destinations = [_cli_endpoint(ctx, command, DST_PREFIX, output_format, output, destination_config_file, dst_fields)] 

326 if _show_preview(ctx, no_preview): 

327 click.echo(render_preview(sources, destinations)) 

328 click.echo(conversion_summary(sources, destinations)) 

329 

330 source_config = _cli_config(source_format, source, source_config_file, src_fields) 

331 destination_config = _cli_config(output_format, output, destination_config_file, dst_fields)

345def _convert_chained( 

346 ctx: click.Context, 

347 sources: list[_ParsedEndpoint], 

348 destinations: list[_ParsedEndpoint], 

349 on_conflict: OnConflict, 

350 fail_fast: bool, 

351 dry_run: bool, 

352 no_preview: bool, 

353) -> None: 

354 """Run the parsed ``source`` / ``destination`` subcommands through :func:`flync_converter.convert_request`. 

355 

356 Args: 

357 ctx: Click context of ``convert``. 

358 sources: The ``source`` occurrences, in merge order. 

359 destinations: The ``destination`` occurrences. 

360 on_conflict: Policy for values two sources disagree on. 

361 fail_fast: Stop at the first destination that fails. 

362 dry_run: Validate the request and the ``--config`` files and print the 

363 preview, without converting (``--dry-run``). 

364 no_preview: Hide the conversion preview (``--no-preview``); ignored by a dry run. 

365 """ 

366 from flync_converter import convert_request

368 def _config(item: _ParsedEndpoint) -> ConverterConfig | str: 

369 """Return the configuration argument of one endpoint, as :func:`_cli_config` combines it.""" 

370 endpoint = item.endpoint 

371 return _cli_config(endpoint.converter_type, endpoint.path, item.config_file, endpoint.options) 

372 

373 try: 

374 request = ConversionRequest( 

375 sources=[ConversionSource(path=s.endpoint.path, converter_type=s.endpoint.converter_type, config=_config(s)) for s in sources], 

376 outputs=[ConversionOutput(path=d.endpoint.path, converter_type=d.endpoint.converter_type, config=_config(d)) for d in destinations], 

377 on_conflict=on_conflict, 

378 fail_fast=fail_fast, 

379 ) 

380 except ValidationError as exc: 

381 raise click.UsageError("; ".join(error["msg"] for error in exc.errors()), ctx) from exc 

382 

383 preview_sources = [item.endpoint for item in sources] 

384 preview_destinations = [item.endpoint for item in destinations] 

385 if dry_run: 

386 _check_config_files(ctx, [*sources, *destinations]) 

387 click.echo(render_preview(preview_sources, preview_destinations)) 

388 click.echo("Dry run: the conversion is valid. Nothing was read or written.") 

389 return 

390 if _show_preview(ctx, no_preview): 

391 click.echo(render_preview(preview_sources, preview_destinations)) 

392 click.echo(conversion_summary(preview_sources, preview_destinations)) 

393 convert_request(request)

396def _check_config_files(ctx: click.Context, items: list[_ParsedEndpoint]) -> None: 

397 """Load every ``--config`` file as its converter's configuration, for ``--dry-run``. 

398 

399 Raises: 

400 click.UsageError: For a file that is not a valid configuration of its converter. 

401 """ 

402 for item in items: 

403 if item.config_file is None: 

404 continue 

405 endpoint = item.endpoint 

406 try: 

407 get_config_model(endpoint.converter_type).from_yaml_file(item.config_file, endpoint.path) 

408 except (ValueError, YAMLError) as exc: 

409 raise click.UsageError(f"--config {item.config_file} of {item.role} {endpoint.path}: {exc}", ctx) from exc

412def _cli_endpoint( 

413 ctx: click.Context, 

414 command: DynamicConverterCommand, 

415 prefix: str, 

416 converter_type: str, 

417 path: str, 

418 config_file: str | None, 

419 fields: dict | None = None, 

420) -> ConversionEndpoint: 

421 """Describe one side of a ``convert`` invocation as a conversion endpoint. 

422 

423 Args: 

424 ctx: Parsed Click context of ``convert`` or of its subcommand. 

425 command: The command holding the injected config options. 

426 prefix: Option prefix of the side (``src_``, ``dst_`` or ``NO_PREFIX``). 

427 converter_type: Converter key of the side. 

428 path: Source or destination path. 

429 config_file: The configuration file given on the command line, if any. 

430 fields: The per-field options given on the command line; read from 

431 ``ctx`` when omitted. 

432 

433 Returns: 

434 The endpoint, with the options exactly as given on the command line. 

435 """ 

436 if fields is None: 

437 fields = { 

438 option.name[len(prefix) :]: ctx.params[option.name] 

439 for option in command._config_options(prefix) 

440 if option.name is not None and ctx.params.get(option.name) is not None 

441 } 

442 arguments = command.build_config_arguments(ctx, prefix) 

443 if config_file is not None: 

444 arguments = ["--config", config_file, *arguments] 

445 return ConversionEndpoint(converter_type, path, fields, arguments)

448def _cli_config(converter_type: str, path: str, config_file: str | None, fields: dict) -> ConverterConfig | str: 

449 """Combine a ``--src-config`` / ``--dst-config`` file and the per-field options into one config argument. 

450 

451 Args: 

452 converter_type: Converter key, used to pick the config class. 

453 path: Source or destination path, used as ``config_path``. 

454 config_file: The configuration file given on the command line, if any. 

455 fields: The per-field options given on the command line. 

456 

457 Returns: 

458 The file path when only a file is given, so it replaces the stored configuration; 

459 the file's configuration with the options applied when both are given; otherwise a 

460 configuration object holding the options, layered over the stored configuration. 

461 """ 

462 model = get_config_model(converter_type) 

463 if config_file is None: 

464 return model(config_path=path, **fields) 

465 if not fields: 

466 return config_file 

467 return model.create_from_config(model.from_yaml_file(config_file, path), **fields)
